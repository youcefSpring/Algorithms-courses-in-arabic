\begin{frame}{أهداف الدرس}
    \begin{tbox}{\faBullseye\quad في نهاية هذا الدرس، تستطيع أن:}
        \begin{itemize}
            \item تعرّف الشجرة الثنائية ومفرداتها: الجذر، الابن، الورقة، الارتفاع.
            \item تمر على شجرة بالطرق الثلاث باستعمال دوال عودية.
            \item تضيف قيمة إلى شجرة بحث ثنائية وتبحث عنها.
        \end{itemize}
    \end{tbox}
\end{frame}

\begin{frame}{مفردات الشجرة}
    \begin{center}
    \begin{tabular}{|c|c|}

        الجذر & العقدة الأولى في الأعلى، ليس لها أب \\ \hline

    \end{tabular}
    \end{center}
\end{frame}

\begin{frame}{قاعدة شجرة البحث}
    \begin{tbox}{\faLightbulb\quad تعريف}
        في \textbf{شجرة البحث الثنائية}: كل قيمة أصغر من العقدة توضع في \textbf{اليسار}، وكل قيمة أكبر منها توضع في \textbf{اليمين}.
    \end{tbox}
\end{frame}

\begin{frame}{طرق المرور على الشجرة السابقة}
    \begin{center}
    \begin{tabular}{|c|c|c|}
        \hline
        \ras{الطريقة} & \ras{الترتيب} & \ras{النتيجة} \\ \hline
        قبلي \textenglish{(preorder)} & العقدة، اليسار، اليمين & \code{50 30 20 40 70 60 80} \\ \hline
        وسطي \textenglish{(inorder)} & اليسار، العقدة، اليمين & \code{20 30 40 50 60 70 80} \\ \hline
        بعدي \textenglish{(postorder)} & اليسار، اليمين، العقدة & \code{20 40 30 60 80 70 50} \\ \hline
    \end{tabular}
    \end{center}
    المرور الوسطي على شجرة بحث يعطي القيم \textbf{مرتبة}.
\end{frame}

\begin{frame}{أخطاء شائعة}
    \begin{obox}{\faExclamationTriangle\quad تجنّب هذه الأخطاء}
        \begin{itemize}
            \item نسيان حالة التوقف \code{t == NULL} في الدوال العودية.
            \item وضع قيمة كبيرة في اليسار: لا تبقى الشجرة شجرة بحث.
            \item تحرير العقدة قبل أبنائها في \code{freeTree}.
        \end{itemize}
    \end{obox}
\end{frame}

\begin{frame}{خلاصة الدرس}
    \begin{tbox}{\faCheckCircle\quad خلاصة الدرس}
        \begin{itemize}
            \item الشجرة الثنائية: لكل عقدة ابن أيسر وابن أيمن على الأكثر.
            \item نعالج الأشجار بدوال عودية: قبلي، وسطي، بعدي.
            \item في شجرة البحث: الأصغر يسارًا والأكبر يمينًا، والمرور الوسطي يعطي ترتيبًا تصاعديًا.
        \end{itemize}
    \end{tbox}
\end{frame}

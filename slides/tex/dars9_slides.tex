\begin{frame}{أهداف الدرس}
    \begin{tbox}{\faBullseye\quad في نهاية هذا الدرس، تستطيع أن:}
        \begin{itemize}
            \item تعرّف هيكلًا بـ\code{struct} وتصل إلى حقوله.
            \item تستعمل جدولًا من الهياكل.
            \item تمرر هيكلًا إلى دالة وتعيده منها.
        \end{itemize}
    \end{tbox}

\begin{frame}{الهيكل}
    \begin{tbox}{\faLightbulb\quad تعريف}

    \end{tbox}
\end{frame}

\begin{frame}{أخطاء شائعة}
    \begin{obox}{\faExclamationTriangle\quad تجنّب هذه الأخطاء}
        \begin{itemize}
            \item نسيان الفاصلة المنقوطة بعد \code{\}} في تعريف الهيكل.
            \item مقارنة هيكلين بـ\code{==}: نقارن الحقول واحدًا واحدًا.
            \item \code{s.name = \"Ali\"}: نستعمل \code{strcpy} للحقل من نوع سلسلة.
        \end{itemize}
    \end{obox}
\end{frame}

\begin{frame}{خلاصة الدرس}
    \begin{tbox}{\faCheckCircle\quad خلاصة الدرس}
        \begin{itemize}
            \item الهيكل يجمع حقولًا مختلفة الأنواع في متغير واحد.
            \item نصل إلى الحقل بالنقطة \code{s.field}، و\code{typedef} تبسط الاسم.
            \item يمكن إنشاء جداول من الهياكل، وتمريرها إلى الدوال وإعادتها منها.
        \end{itemize}
    \end{tbox}
\end{frame}

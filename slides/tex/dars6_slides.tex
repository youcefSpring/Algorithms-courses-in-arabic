\begin{frame}{دوال المكتبة \textenglish{string.h}}
    \begin{center}
    \begin{tabular}{|c|c|}
        \hline
        \ras{الدالة} & \ras{دورها} \\ \hline
        \code{strlen(s)} & طول السلسلة دون \code{'\textbackslash 0'} \\ \hline
        \code{strcpy(d, s)} & نسخ \code{s} في \code{d} \\ \hline
        \code{strcat(d, s)} & إلحاق \code{s} بنهاية \code{d} \\ \hline
        \code{strcmp(a, b)} & مقارنة: \textenglish{0} إذا كانتا متساويتين \\ \hline
    \end{tabular}
    \end{center}
    نضيف \code{\#include <string.h>} في بداية البرنامج.
\end{frame}

\begin{frame}{أخطاء شائعة}
    \begin{obox}{\faExclamationTriangle\quad تجنّب هذه الأخطاء}
        \begin{itemize}
            \item نسيان الخانة الإضافية للحرف \code{'\textbackslash 0'}.
            \item استعمال \code{=} بدل \code{strcpy}، أو \code{==} بدل \code{strcmp}.
            \item وضع \code{\&} قبل اسم السلسلة في \code{scanf}.
        \end{itemize}
    \end{obox}
\end{frame}

\begin{frame}{خلاصة الدرس}
    \begin{tbox}{\faCheckCircle\quad خلاصة الدرس}
        \begin{itemize}
            \item السلسلة جدول حروف ينتهي بـ\code{'\textbackslash 0'}.
            \item نمر على الحروف بحلقة حتى \code{'\textbackslash 0'}.
            \item \code{string.h}: \code{strlen}، \code{strcpy}، \code{strcat}، \code{strcmp}.
        \end{itemize}
    \end{tbox}

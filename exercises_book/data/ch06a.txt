- إذا كان الحرف حرفًا أو رقمًا: نكتبه صغيرًا.

if: write > 0 && slug[write-1] == '-'

out: slug
@ code
#include <stdio.h>

int main(void) {
    char name[] = "Smart TV 55\" (4K) -- Ultra HD!";
    char slug[60], ch;
    int read, write = 0;
    for (read = 0; name[read] != '\0'; read++) {
        ch = name[read];
        if (ch >= 'A' && ch <= 'Z')
            ch += 'a' - 'A';                  // حروف صغيرة

            slug[write++] = '-';              // فاصل واحد فقط

    if (write > 0 && slug[write - 1] == '-')
        write--;
    slug[write] = '\0';
    printf("https://shop.dz/p/%s\n", slug);
    return 0;
}

@ steps
- نبحث عن الكلمة في كل موضع.
- نتحقق أن الحرف قبلها وبعدها ليس حرفًا (حدود كلمة).

@ flow
for: start = 0 ; text[start] != '\0' ; start++
  do: match = طول التطابق في start
  if: تطابق كامل && حدود كلمة
    for: pos = start ; pos < start + match ; pos++
      do: text[pos] = '*'
out: text
@ code
#include <stdio.h>

int main(void) {
    char text[] = "This seller is a scam, total scam! scampi is fine.";
    char banned[] = "scam";
    int start, match, pos, before, after;
    for (start = 0; text[start] != '\0'; start++) {
        match = 0;
        while (banned[match] != '\0' && text[start + match] == banned[match])
            match++;
        if (banned[match] != '\0')

            for (pos = start; pos < start + match; pos++)
                text[pos] = '*';
    }
    printf("%s\n", text);
    return 0;
}

title: تشفير رمز القسيمة بإزاحة قيصر
level: medium
domain: ec
@ statement
لإخفاء رموز القسائم في الرسائل التسويقية نستعمل **تشفير قيصر**: كل حرف يُزاح بـ `shift` موضعًا في الأبجدية (مع الدوران بعد Z)، والأرقام تُزاح في 0..9. اكتب برنامجًا يشفر ثم يفك التشفير للتحقق.
@ stdin
SALE2026 3
@ steps
- حرف كبير: `'A' + (ch - 'A' + shift) % 26`.
- رقم: `'0' + (ch - '0' + shift) % 10`.
- فك التشفير: إزاحة بـ `26 - shift` و`10 - shift % 10`.
@ flow
in: code, shift
for: pos = 0 ; code[pos] != '\0' ; pos++
  if: حرف كبير
    do: code[pos] = 'A' + (code[pos] - 'A' + shift) % 26
  else:
    if: رقم
      do: code[pos] = '0' + (code[pos] - '0' + shift) % 10
out: code
@ code
#include <stdio.h>

int main(void) {
    char code[30];
    int shift, pos, round, step;
    scanf("%29s %d", code, &shift);
    for (round = 0; round < 2; round++) {
        // الجولة 0: تشفير، الجولة 1: فك التشفير
        step = (round == 0) ? shift : 26 - shift % 26;
        for (pos = 0; code[pos] != '\0'; pos++) {
            if (code[pos] >= 'A' && code[pos] <= 'Z')
                code[pos] = 'A' + (code[pos] - 'A' + step) % 26;   // دوران
            else if (code[pos] >= '0' && code[pos] <= '9')
                code[pos] = '0' + (code[pos] - '0' + (round == 0 ? shift : 10 - shift % 10)) % 10;
        }
        printf("%s: %s\n", round == 0 ? "Encrypted" : "Decrypted", code);
    }
    return 0;
}

@ code
#include <stdio.h>

#define NB 8

typedef struct { int start, end; } Window;

int main(void) {
    Window windows[NB] = {{14, 16}, {9, 11}, {10, 12}, {18, 20}, {15, 17}, {8, 9}, {19, 21}, {13, 14}};
    Window key, current;
    int idx, back, busy = 0;
    for (idx = 1; idx < NB; idx++) {                 // ترتيب حسب البداية
        key = windows[idx];
        for (back = idx - 1; back >= 0 && windows[back].start > key.start; back--)
            windows[back + 1] = windows[back];
        windows[back + 1] = key;
    }
    current = windows[0];
    for (idx = 1; idx < NB; idx++) {
        if (windows[idx].start <= current.end) {     // تداخل أو تلامس
            if (windows[idx].end > current.end) current.end = windows[idx].end;
        } else {
            printf("Shift %02dh-%02dh\n", current.start, current.end);
            busy += current.end - current.start;
            current = windows[idx];
        }
    }
    printf("Shift %02dh-%02dh\n", current.start, current.end);
    busy += current.end - current.start;
    printf("Driver works %d hours\n", busy);

typedef struct {
    char name[12];
    double posX, posY;
    int orders;
} City;

typedef struct { double posX, posY; } Point;

double mySqrt(double value) {
    double guess = value > 1 ? value : 1;
    int iter;
    for (iter = 0; iter < 50; iter++) guess = (guess + value / guess) / 2;
    return guess;
}

        sumX += cities[idx].orders * cities[idx].posX;

#include <stdio.h>

typedef struct { int day, month, year; } Date;

int daysIn(int month, int year) {
    int days[12] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    if (month == 2 && ((year % 4 == 0 && year % 100 != 0) || year % 400 == 0)) return 29;
    return days[month - 1];
}

Date addDays(Date date, int count) {
    while (count-- > 0) {
        date.day++;
        if (date.day > daysIn(date.month, date.year)) {   // شهر جديد
            date.day = 1;
            date.month++;
            if (date.month > 12) { date.month = 1; date.year++; }
        }
    }
    return date;
}

    int idx;
    printf("Purchase: %02d/%02d/%d\n", purchase.day, purchase.month, purchase.year);
    for (idx = 1; idx <= 4; idx++) {
        due = addDays(due, 30);
        printf("Installment %d due %02d/%02d/%d\n", idx, due.day, due.month, due.year);
    }
    return 0;

#include <stdio.h>

typedef struct { int code; double price; int qty; } CartItem;
typedef struct { CartItem items[10]; int count; } Cart;

void addItem(Cart cart[], int code, double price, int qty) {
    int idx;
    for (idx = 0; idx < cart[0].count; idx++)
        if (cart[0].items[idx].code == code) { cart[0].items[idx].qty += qty; return; }
    cart[0].items[cart[0].count].code = code;           // سطر جديد
    cart[0].items[cart[0].count].price = price;
    cart[0].items[cart[0].count].qty = qty;
    cart[0].count++;
}

void removeItem(Cart cart[], int code) {
    int idx, found = -1;
    for (idx = 0; idx < cart[0].count; idx++)
        if (cart[0].items[idx].code == code) found = idx;
    if (found < 0) return;
    for (idx = found; idx < cart[0].count - 1; idx++)
        cart[0].items[idx] = cart[0].items[idx + 1];      // إزاحة
    cart[0].count--;
}

double cartTotal(Cart cart[]) {
    double total = 0;
    int idx;
    for (idx = 0; idx < cart[0].count; idx++)
        total += cart[0].items[idx].price * cart[0].items[idx].qty;
    return total;
}

void show(Cart cart[]) {
    int idx;
    for (idx = 0; idx < cart[0].count; idx++)
        printf("[%d x%d] ", cart[0].items[idx].code, cart[0].items[idx].qty);
    printf("= %.2f DA\n", cartTotal(cart));
}

    Cart cart[1];

    addItem(cart, 330, 899, 3);  show(cart);

#define NB 20
#define NBB 5

typedef struct { double low, high; int count; } Bucket;

int main(void) {

                         2700, 450, 6100, 3900, 2100, 1800, 4200, 3600, 2900, 5100};

    int idx, bin, star;

        if (values[idx] > high) high = values[idx];

        printf("%6.0f-%6.0f | ", buckets[bin].low, buckets[bin].high);

title: منطقة تغطية التوصيلات
level: medium
domain: ec
@ statement
عرّف هيكل `Rect` (الزاوية الدنيا والعليا) ودالة `boundingBox(points, size)` تعيد أصغر مستطيل يحتوي كل نقاط التسليم لسائق، ودالة `overlap(first, second)` تحسب مساحة تقاطع منطقتي سائقين (لتفادي التداخل).
@ steps
- المستطيل المحيط: أصغر وأكبر `posX` و `posY`.
- التقاطع: `width = min(highs) - max(lows)` في كل محور؛ إذا كان سالبًا فلا تقاطع.
@ flow
call: zoneA = boundingBox(driverA, 5)
call: zoneB = boundingBox(driverB, 4)

out: zoneA, zoneB, shared
@ code
#include <stdio.h>

typedef struct { double posX, posY; } Point;
typedef struct { Point low, high; } Rect;

Rect boundingBox(Point points[], int size) {
    Rect box;
    int idx;
    box.low = box.high = points[0];
    for (idx = 1; idx < size; idx++) {
        if (points[idx].posX < box.low.posX) box.low.posX = points[idx].posX;
        if (points[idx].posY < box.low.posY) box.low.posY = points[idx].posY;
        if (points[idx].posX > box.high.posX) box.high.posX = points[idx].posX;
        if (points[idx].posY > box.high.posY) box.high.posY = points[idx].posY;
    }
    return box;
}

double overlap(Rect first, Rect second) {
    double left = first.low.posX > second.low.posX ? first.low.posX : second.low.posX;
    double right = first.high.posX < second.high.posX ? first.high.posX : second.high.posX;
    double bottom = first.low.posY > second.low.posY ? first.low.posY : second.low.posY;
    double top = first.high.posY < second.high.posY ? first.high.posY : second.high.posY;
    if (right <= left || top <= bottom) return 0;           // لا تقاطع
    return (right - left) * (top - bottom);
}

@ steps
- المنتج يُقبل إذا حقق **كل** شروط المرشح.
- الشرط `inStockOnly` يُطبق فقط إذا كانت قيمته 1.
@ flow

  for: idx = 0 ; idx < 8 ; idx++

@ code
#include <stdio.h>

typedef struct { char name[14]; char category; double price, rating; int stock; } Product;
typedef struct { char category; double minPrice, maxPrice, minRating; int inStockOnly; } Filter;

int matches(Product item, Filter filter) {
    if (filter.category != '*' && item.category != filter.category) return 0;
    if (item.price < filter.minPrice || item.price > filter.maxPrice) return 0;
    if (item.rating < filter.minRating) return 0;
    if (filter.inStockOnly && item.stock == 0) return 0;
    return 1;                                        // كل الشروط محققة
}

int main(void) {
    Product items[8] = {{"Galaxy A", 'P', 38000, 4.3, 5}, {"Redmi 13", 'P', 27500, 4.1, 0},
                        {"iPhone", 'P', 165000, 4.7, 2}, {"ThinkPad", 'L', 95000, 4.5, 3},
                        {"IdeaPad", 'L', 72000, 3.9, 6}, {"Earbuds", 'A', 5400, 4.6, 20},
                        {"Charger", 'A', 2200, 3.5, 0}, {"Hub", 'A', 3100, 4.2, 8}};
    Filter filters[3] = {{'P', 0, 50000, 4.0, 0}, {'*', 3000, 100000, 4.2, 1}, {'A', 0, 10000, 0, 1}};
    int filt, idx;
    for (filt = 0; filt < 3; filt++) {
        printf("Filter %d:", filt + 1);
        for (idx = 0; idx < 8; idx++)
            if (matches(items[idx], filters[filt])) printf(" %s;", items[idx].name);
        printf("\n");
    }
    return 0;
}

level: medium

                           {6, 'R', 1200, 0}, {9, 'P', 6200, 0}, {12, 'D', 5000, 0},
                           {15, 'P', 3300, 0}, {20, 'P', 1800, 0}};

        }

               log[idx].amount, log[idx].accepted ? "ok" : "REFUSED", balance);
    }
    printf("Lowest balance: %.2f DA\n", lowest);

}

title: اختيار المورد بالعلامة المرجحة
level: medium
domain: ds
@ statement
لكل مورد هيكل يحتوي على: السعر، مدة التسليم بالأيام، نسبة العيوب، والتقييم. نطبّع كل معيار إلى `[0, 1]` (الأفضل = 1)، ثم نحسب علامة مرجحة: السعر 40%، التسليم 25%، العيوب 20%، التقييم 15%. اعرض الترتيب.
@ steps
- للمعايير التي "الأقل أفضل" (السعر، التسليم، العيوب): `(max - value) / (max - min)`.
- للتقييم: `(value - min) / (max - min)`.
- العلامة = مجموع الأوزان × العلامات المطبّعة.
@ flow
for: crit = 0 ; crit < 4 ; crit++
  do: إيجاد min و max للمعيار
for: idx = 0 ; idx < 4 ; idx++
  do: score = Σ weight[crit] * normalized(idx, crit)
out: الموردون مرتبون حسب score
@ code
#include <stdio.h>

#define NB 4

typedef struct { char name[10]; double values[4]; double score; } Supplier;

int main(void) {
    Supplier list[NB] = {{"AlphaTec", {1200, 5, 2.0, 4.1}, 0}, {"BetaCom", {1050, 9, 3.5, 3.8}, 0},
                         {"Gamma", {1350, 3, 0.8, 4.6}, 0}, {"Delta", {1100, 6, 1.5, 4.0}, 0}};
    double weights[4] = {0.40, 0.25, 0.20, 0.15};
    int lowerBetter[4] = {1, 1, 1, 0};
    double low, high, norm;
    Supplier tmp;
    int crit, idx, pass;
    for (crit = 0; crit < 4; crit++) {
        low = high = list[0].values[crit];
        for (idx = 1; idx < NB; idx++) {
            if (list[idx].values[crit] < low) low = list[idx].values[crit];
            if (list[idx].values[crit] > high) high = list[idx].values[crit];
        }
        for (idx = 0; idx < NB; idx++) {
            norm = lowerBetter[crit] ? (high - list[idx].values[crit]) / (high - low)
                                     : (list[idx].values[crit] - low) / (high - low);
            list[idx].score += weights[crit] * norm;          // مساهمة المعيار
        }
    }
    for (pass = 0; pass < NB - 1; pass++)
        for (idx = 0; idx < NB - 1 - pass; idx++)
            if (list[idx].score < list[idx + 1].score) {
                tmp = list[idx]; list[idx] = list[idx + 1]; list[idx + 1] = tmp;
            }
    for (idx = 0; idx < NB; idx++)
        printf("%d. %-9s score %.3f\n", idx + 1, list[idx].name, list[idx].score);
    return 0;
}

level: medium
domain: ds
@ statement
لكل منتج هيكل يحتوي على اسمه وجدول أسعاره خلال 10 أسابيع. اكتب دالة `largestDrop(item, from, to)` تجد **أكبر انخفاض** بين أسبوع وأسبوع لاحق (ليس بالضرورة متتاليين) في مرور واحد، وتكتب الأسبوعين في جدول.
@ steps
- نحتفظ بأعلى سعر سابق `peak` وأسبوعه.
- لكل أسبوع: الانخفاض = `peak - price`؛ إذا كان أكبر من الأفضل نحفظه.
- مرور واحد: تعقيد خطي بدل مقارنة كل الأزواج.
@ flow
do: peak = prices[0] ; best = 0
for: week = 1 ; week < 10 ; week++
  if: peak - prices[week] > best
    do: best = peak - prices[week] ; from = peakWeek ; to = week
  if: prices[week] > peak
    do: peak = prices[week] ; peakWeek = week
out: best, from, to
@ code
#include <stdio.h>

typedef struct { char name[12]; double prices[10]; } History;

double largestDrop(History item, int weeks[]) {
    double peak = item.prices[0], best = 0;
    int week, peakWeek = 0;
    weeks[0] = weeks[1] = 0;
    for (week = 1; week < 10; week++) {
        if (peak - item.prices[week] > best) {
            best = peak - item.prices[week];
            weeks[0] = peakWeek;
            weeks[1] = week;
        }
        if (item.prices[week] > peak) { peak = item.prices[week]; peakWeek = week; }   // قمة جديدة
    }
    return best;
}

int main(void) {
    History items[3] = {{"Phone", {32000, 33500, 31000, 34500, 30000, 29000, 31500, 28000, 29500, 30500}},
                        {"Laptop", {89000, 87000, 91000, 90500, 88000, 92000, 95000, 94000, 93000, 96000}},
                        {"Earbuds", {5400, 5200, 5000, 4800, 4600, 4500, 4400, 4300, 4200, 4100}}};
    int idx, weeks[2];
    double drop;
    for (idx = 0; idx < 3; idx++) {
        drop = largestDrop(items[idx], weeks);
        printf("%-8s largest drop %6.0f DA (week %d -> week %d, -%.1f%%)\n", items[idx].name,
               drop, weeks[0] + 1, weeks[1] + 1, 100 * drop / items[idx].prices[weeks[0]]);
    }
    return 0;
}

level: medium
domain: ec
@ statement
إصدار التطبيق يُكتب `major.minor.patch` مثل `2.10.3`. اقرأ الإصدارات كنصوص، وحوّلها يدويًا إلى هيكل `Version`، واكتب دالة `compareVersions`، ثم حدد لكل جهاز هل يحتاج إلى تحديث نحو الإصدار الأخير `2.10.3`، مع التنبيه إذا كان التحديث **رئيسيًا**.
@ stdin
4
2.9.12
2.10.3
1.15.0
2.10.1
@ steps
- التحويل: نقرأ الأرقام حرفًا بحرف ونمر إلى الحقل التالي عند كل نقطة.
- المقارنة: `major` ثم `minor` ثم `patch` (المقارنة النصية خاطئة: `"2.9" > "2.10"` نصيًا).
@ flow
in: nb
for: idx = 0 ; idx < nb ; idx++
  in: text
  call: device = parseVersion(text)
  if: compareVersions(device, latest) < 0
    out: `"update"`, رئيسي إذا اختلف major
  else:
    out: `"up to date"`
@ code
#include <stdio.h>

typedef struct { int major, minor, patch; } Version;

Version parseVersion(char text[]) {
    int parts[3] = {0, 0, 0}, field = 0, pos;
    Version result;
    for (pos = 0; text[pos] != '\0' && field < 3; pos++) {
        if (text[pos] == '.') field++;                       // الحقل التالي
        else parts[field] = parts[field] * 10 + (text[pos] - '0');
    }
    result.major = parts[0];
    result.minor = parts[1];
    result.patch = parts[2];
    return result;
}

int compareVersions(Version first, Version second) {
    if (first.major != second.major) return first.major - second.major;
    if (first.minor != second.minor) return first.minor - second.minor;
    return first.patch - second.patch;
}

int main(void) {
    Version latest = {2, 10, 3}, device;
    char text[20];
    int nb, idx;
    scanf("%d", &nb);
    for (idx = 0; idx < nb; idx++) {
        scanf("%19s", text);
        device = parseVersion(text);
        if (compareVersions(device, latest) < 0)
            printf("%-8s -> update to 2.10.3%s\n", text, device.major < latest.major ? " (MAJOR)" : "");
        else
            printf("%-8s -> up to date\n", text);
    }
    return 0;
}

title: نتائج اختبار A/B لصفحة المنتج
level: medium
domain: ds
@ statement
اختبر المتجر نسختين من صفحة المنتج. لكل نسخة هيكل `Variant` (الاسم، عدد الزوار، عدد المشترين). احسب معدل التحويل لكل نسخة، و**الارتفاع النسبي** (lift)، وإحصائية `z` لاختبار الفرق بين نسبتين، وقرر إن كان الفرق معنويًا (`|z| > 1.96`).
@ steps
- `rate = buyers / visitors`؛ `lift = (rateB - rateA) / rateA`.
- النسبة المجمعة `pooled = (buyersA + buyersB) / (visitorsA + visitorsB)`.
- `z = (rateB - rateA) / sqrt(pooled (1 - pooled) (1/nA + 1/nB))`.
@ flow
do: rateA = buyersA / visitorsA ; rateB = buyersB / visitorsB
do: pooled = (buyersA + buyersB) / (visitorsA + visitorsB)
do: z = (rateB - rateA) / sqrt(pooled (1 - pooled) (1/nA + 1/nB))
if: |z| > 1.96
  out: `"significant"`
else:
  out: `"not significant"`
@ code
#include <stdio.h>

typedef struct { char name[16]; int visitors, buyers; } Variant;

double mySqrt(double value) {
    double guess = value > 1 ? value : 1;
    int iter;
    for (iter = 0; iter < 60; iter++) guess = (guess + value / guess) / 2;
    return guess;
}

double rate(Variant variant) { return (double)variant.buyers / variant.visitors; }

int main(void) {
    Variant tests[2][2] = {{{"A: old page", 4800, 192}, {"B: new photos", 4750, 238}},
                           {{"A: green btn", 1200, 54}, {"B: red btn", 1180, 59}}};
    double rateA, rateB, pooled, zValue;
    int test;
    for (test = 0; test < 2; test++) {
        rateA = rate(tests[test][0]);
        rateB = rate(tests[test][1]);
        pooled = (double)(tests[test][0].buyers + tests[test][1].buyers)
               / (tests[test][0].visitors + tests[test][1].visitors);
        zValue = (rateB - rateA) / mySqrt(pooled * (1 - pooled) *
                 (1.0 / tests[test][0].visitors + 1.0 / tests[test][1].visitors));
        printf("%s %.2f%% vs %s %.2f%%\n", tests[test][0].name, 100 * rateA,
               tests[test][1].name, 100 * rateB);
        printf("  lift %+.1f%%, z = %.2f -> %s\n", 100 * (rateB - rateA) / rateA, zValue,
               (zValue > 1.96 || zValue < -1.96) ? "significant" : "not significant");
    }
    return 0;
}

typedef struct {
    char name[20];
    char category;
    double price;
} Product;

int cheapestOf(Product items[], int size, char category) {
    int idx, best = -1;
    for (idx = 0; idx < size; idx++)
        if (items[idx].category == category &&
            (best < 0 || items[idx].price < items[best].price))
            best = idx;
    return best;
}

        printf("Category %c: %-9s %9.2f DA\n", categories[cat], items[best].name, items[best].price);

@ steps
- `add`: `a/b + c/d = (ad + cb) / bd` ثم التبسيط بالقاسم المشترك الأكبر.
- `multiply`: `(a·c) / (b·d)`.
- الباقي = `1 - (1/3 + 1/4)`.
@ flow
call: taken = add(first, second)
do: rest = 1 - taken
call: third = multiply(rest, 1/3)
out: first, second, third
@ code
#include <stdio.h>

typedef struct {
    long num, den;
} Fraction;

long gcd(long first, long second) {
    while (second != 0) { long rest = first % second; first = second; second = rest; }
    return first < 0 ? -first : first;
}

Fraction simplify(Fraction value) {
    long divisor = gcd(value.num, value.den);
    value.num /= divisor;
    value.den /= divisor;
    return value;
}

Fraction add(Fraction first, Fraction second) {
    Fraction sum = {first.num * second.den + second.num * first.den, first.den * second.den};
    return simplify(sum);
}

Fraction multiply(Fraction first, Fraction second) {
    Fraction product = {first.num * second.num, first.den * second.den};
    return simplify(product);
}

int main(void) {
    Fraction first = {1, 3}, second = {1, 4}, oneThird = {1, 3};
    Fraction taken = add(first, second);
    Fraction rest = {taken.den - taken.num, taken.den};     // 1 - taken
    Fraction third = multiply(rest, oneThird);
    Fraction left = add(taken, third);
    printf("Partner 1: %ld/%ld\nPartner 2: %ld/%ld\nPartner 3: %ld/%ld\n",
           first.num, first.den, second.num, second.den, third.num, third.den);
    printf("Reinvested: %ld/%ld of profit\n", left.den - left.num, left.den);
    return 0;
}

@ steps
- دالة `sortDims` ترتب أبعاد الصندوق تصاعديًا (3 مقارنات وتبديلات).
- يدخل إذا كان كل بعد مرتب للمنتج ≤ البعد المرتب الموافق للصندوق.
- أصغر صندوق مناسب = أصغر حجم بين الصناديق المناسبة.
@ flow

  do: best = -1
  for: carton = 0 ; carton < 4 ; carton++
    if: fitsIn(items[item], cartons[carton]) && حجمه أصغر

@ code
#include <stdio.h>

typedef struct {
    double length, width, height;
} Box;

Box sortDims(Box box) {
    double tmp;
    if (box.length > box.width)  { tmp = box.length; box.length = box.width; box.width = tmp; }
    if (box.width > box.height)  { tmp = box.width; box.width = box.height; box.height = tmp; }
    if (box.length > box.width)  { tmp = box.length; box.length = box.width; box.width = tmp; }
    return box;                    // length <= width <= height
}

double volume(Box box) { return box.length * box.width * box.height; }

int fitsIn(Box item, Box carton) {
    item = sortDims(item);
    carton = sortDims(carton);
    return item.length <= carton.length && item.width <= carton.width && item.height <= carton.height;
}

int main(void) {
    Box cartons[4] = {{30, 20, 10}, {40, 30, 20}, {15, 60, 15}, {50, 40, 40}};
    Box items[3] = {{12, 55, 9}, {35, 18, 25}, {28, 8, 19}};
    int item, carton, best;
    for (item = 0; item < 3; item++) {
        best = -1;
        for (carton = 0; carton < 4; carton++)
            if (fitsIn(items[item], cartons[carton]) &&
                (best < 0 || volume(cartons[carton]) < volume(cartons[best])))
                best = carton;
        printf("Item %d (%.0fx%.0fx%.0f): carton %d (%.0f cm3)\n", item + 1, items[item].length,
               items[item].width, items[item].height, best + 1, volume(cartons[best]));

@ statement
الدالة في C تعيد قيمة واحدة، لكن يمكنها أن تعيد **هيكلًا** يحتوي على عدة نتائج. اكتب دالة `describe(values, size)` تعيد هيكل `Summary` (العدد، الأصغر، الأكبر، المتوسط، الانحراف المعياري) لمبيعات فرعين.
@ steps
- مرور أول: الأصغر والأكبر والمجموع؛ مرور ثانٍ: مجموع مربعات الانحرافات.
- الجذر بطريقة نيوتن.

#include <stdio.h>

typedef struct {
    int count;
    double low, high, mean, std;
} Summary;

double mySqrt(double value) {
    double guess = value > 1 ? value : 1;
    int iter;
    for (iter = 0; iter < 50; iter++) guess = (guess + value / guess) / 2;
    return guess;
}

Summary describe(double values[], int size) {
    Summary result;
    double sum = 0, squares = 0;
    int idx;
    result.count = size;
    result.low = result.high = values[0];
    for (idx = 0; idx < size; idx++) {
        sum += values[idx];
        if (values[idx] < result.low) result.low = values[idx];
        if (values[idx] > result.high) result.high = values[idx];
    }
    result.mean = sum / size;
    for (idx = 0; idx < size; idx++)
        squares += (values[idx] - result.mean) * (values[idx] - result.mean);
    result.std = mySqrt(squares / size);
    return result;                        // عدة نتائج في هيكل واحد
}

void printSummary(char label[], Summary info) {
    printf("%s n=%d min=%.0f max=%.0f mean=%.2f std=%.2f\n", label, info.count,
           info.low, info.high, info.mean, info.std);
}

int main(void) {
    double salesNorth[7] = {120, 135, 128, 150, 142, 160, 131};
    double salesSouth[7] = {90, 210, 60, 180, 75, 200, 95};
    printSummary("North:", describe(salesNorth, 7));
    printSummary("South:", describe(salesSouth, 7));
    return 0;
}

int countChars(const char *path) {
    FILE *file = fopen(path, "r");
    int count = 0;
    if (file == NULL) return -1;
    while (fgetc(file) != EOF) count++;
    fclose(file);
    return count;
}

int main(void) {

    source = fopen("customers.txt", "w");

    fclose(target);

#include <stdio.h>

typedef struct { char name[16]; double revenue; } Total;

int toInt(const char *text) {
    int value = 0;
    while (*text >= '0' && *text <= '9') value = value * 10 + (*text++ - '0');
    return value;
}

double toDouble(const char *text) {
    double value = 0, scale = 1;
    while (*text >= '0' && *text <= '9') value = value * 10 + (*text++ - '0');
    if (*text == '.') for (text++; *text >= '0' && *text <= '9'; text++) value += (*text - '0') * (scale /= 10);
    return value;
}

int sameText(const char *first, const char *second) {
    while (*first != '\0' && *first == *second) { first++; second++; }
    return *first == *second;
}

int main(void) {
    FILE *file = fopen("sales.csv", "w");
    char line[100], *fields[4];
    Total totals[10];
    int nbTotals = 0, count, idx, pos, letter;
    fprintf(file, "2026-11-01;Mouse;3;1500\n2026-11-01;Cable;10;450.5\n2026-11-02;Mouse;2;1450\n");
    fprintf(file, "2026-11-02;Monitor;1;32000\n2026-11-03;Cable;4;450.5\n2026-11-03;Mouse;1;1500\n");
    fclose(file);
    file = fopen("sales.csv", "r");
    if (file == NULL) return 1;
    while (fgets(line, 100, file) != NULL) {
        count = 1;
        fields[0] = line;
        for (pos = 0; line[pos] != '\0' && line[pos] != '\n'; pos++)
            if (line[pos] == ';' && count < 4) { line[pos] = '\0'; fields[count++] = line + pos + 1; }
        line[pos] = '\0';
        for (idx = 0; idx < nbTotals && !sameText(totals[idx].name, fields[1]); idx++) {}
        if (idx == nbTotals) {                                   // منتج جديد
            for (letter = 0; (totals[idx].name[letter] = fields[1][letter]) != '\0'; letter++) {}
            totals[idx].revenue = 0;
            nbTotals++;
        }
        totals[idx].revenue += toInt(fields[2]) * toDouble(fields[3]);
    }
    fclose(file);
    for (idx = 0; idx < nbTotals; idx++) printf("%-8s %10.2f DA\n", totals[idx].name, totals[idx].revenue);
    return 0;
}

        for (col = 0; col < 3; col++) {
            sum[col] += row[col];
            sumSq[col] += row[col] * row[col];
            if (count == 0 || row[col] < low[col]) low[col] = row[col];
            if (count == 0 || row[col] > high[col]) high[col] = row[col];
        }
        count++;
    }
    fclose(file);
    for (col = 0; col < 3; col++) {
        mean = sum[col] / count;
        printf("%-8s mean %10.2f  std %9.2f  min %8.0f  max %8.0f\n", names[col], mean,
               mySqrt(sumSq[col] / count - mean * mean), low[col], high[col]);
    }
    return 0;
}

@ statement
تطلب واجهة الهاتف الكتالوج بصيغة **JSON**. اكتب دالة `exportJson(path, items, size)` تكتب مصفوفة كائنات مع الفواصل والمسافات الصحيحة (لا فاصلة بعد آخر عنصر)، و**تهرّب** علامات الاقتباس داخل الأسماء، ثم اعرض الملف.
@ steps
- `[` ثم لكل منتج `{"code": ..., "name": "...", "price": ...}` ثم `]`.
- الفاصلة بين العناصر فقط: `if (idx < size - 1)`.
- الحرف `"` داخل الاسم يُكتب `\"`.
@ flow
do: fprintf("[\n")
for: idx = 0 ; idx < size ; idx++
  do: كتابة الكائن مع تهريب الاسم
  if: idx < size - 1
    do: fprintf(",")
do: fprintf("]\n")
@ code
#include <stdio.h>

typedef struct { int code; char name[30]; double price; int inStock; } Product;

void writeEscaped(FILE *file, const char *text) {
    for (; *text != '\0'; text++) {
        if (*text == '"' || *text == '\\') fputc('\\', file);   // تهريب
        fputc(*text, file);
    }
}

void exportJson(const char *path, const Product *items, int size) {
    FILE *file = fopen(path, "w");
    int idx;
    fprintf(file, "[\n");
    for (idx = 0; idx < size; idx++) {
        fprintf(file, "  {\"code\": %d, \"name\": \"", items[idx].code);
        writeEscaped(file, items[idx].name);
        fprintf(file, "\", \"price\": %.2f, \"inStock\": %s}", items[idx].price,
                items[idx].inStock ? "true" : "false");
        if (idx < size - 1) fprintf(file, ",");                   // لا فاصلة بعد الأخير
        fprintf(file, "\n");
    }
    fprintf(file, "]\n");
    fclose(file);
}

int main(void) {
    Product items[3] = {{101, "Mouse", 1500, 1}, {205, "Monitor 27\" 4K", 54000, 1}, {330, "Hub", 3100, 0}};
    FILE *file;
    int ch;
    exportJson("catalog.json", items, 3);
    file = fopen("catalog.json", "r");
    while ((ch = fgetc(file)) != EOF) putchar(ch);
    fclose(file);
    return 0;
}

title: كشف البريد المكرر في ملف الزبائن
level: medium
domain: ds
@ statement
ملف التسجيلات `signups.txt` يحتوي على بريد إلكتروني في كل سطر، بعضها مكرر بحروف مختلفة الحجم (`Amina@Mail.dz` و `amina@mail.dz`). اقرأ الملف، وحّد كل بريد إلى الحروف الصغيرة، واكتب البريد الفريد فقط في `unique.txt` مع عدّ المكررات.
@ steps
- `normalize`: تحويل إلى حروف صغيرة في مكانها.
- نحتفظ بالبريد المرئي سابقًا في جدول ونبحث فيه.
- نكتب في الملف الجديد عند أول ظهور فقط.
@ flow
while: fscanf(input, email) == 1
  call: normalize(email)
  if: email غير موجود في seen
    do: إضافة إلى seen ; fprintf(output, email)
  else:
    do: duplicates++

@ code
#include <stdio.h>

#define MAXE 50

void normalize(char *text) {
    for (; *text != '\0'; text++)
        if (*text >= 'A' && *text <= 'Z') *text = *text - 'A' + 'a';
}

int sameText(const char *first, const char *second) {
    while (*first != '\0' && *first == *second) { first++; second++; }
    return *first == *second;
}

int main(void) {
    FILE *input = fopen("signups.txt", "w"), *output;
    char seen[MAXE][40], email[40];
    int nb = 0, duplicates = 0, idx, letter;
    fprintf(input, "amina@mail.dz\nkarim@shop.dz\nAmina@Mail.dz\nlina@web.dz\nKARIM@shop.dz\n");
    fprintf(input, "omar@mail.dz\namina@mail.dz\n");
    fclose(input);
    input = fopen("signups.txt", "r");
    output = fopen("unique.txt", "w");
    while (fscanf(input, "%39s", email) == 1) {
        normalize(email);
        for (idx = 0; idx < nb && !sameText(seen[idx], email); idx++) {}
        if (idx == nb && nb < MAXE) {                     // أول ظهور
            for (letter = 0; (seen[nb][letter] = email[letter]) != '\0'; letter++) {}
            nb++;
            fprintf(output, "%s\n", email);
        } else {
            duplicates++;
        }
    }
    fclose(input);
    fclose(output);
    printf("%d unique, %d duplicates removed\n", nb, duplicates);
    output = fopen("unique.txt", "r");
    while (fscanf(output, "%39s", email) == 1) printf("  %s\n", email);
    fclose(output);
    return 0;
}

title: مدرج التقييمات من ملف
level: medium
domain: ds
@ statement
ملف `ratings.txt` يحتوي على تقييمات من 1 إلى 5 (قيمة في كل سطر)، وقد يحتوي على قيم غير صالحة (0، 7، أو نص). اقرأ الملف، وتجاهل القيم غير الصالحة مع عدّها، ثم اكتب تقريرًا في `report.txt` (المدرج بالنجوم والمتوسط) واعرضه.
@ steps
- إذا فشل `fscanf` في قراءة عدد (نص)، نتخطى الكلمة بـ `fscanf(file, "%*s")`.
- `counts[value]` لـ `value` من 1 إلى 5.
@ flow
while: لم نصل إلى EOF
  if: fscanf(file, value) == 1
    if: 1 <= value <= 5
      do: counts[value]++
    else:
      do: invalid++
  else:
    do: تخطي كلمة غير رقمية ; invalid++
do: كتابة التقرير في report.txt
@ code
#include <stdio.h>

int main(void) {
    FILE *file = fopen("ratings.txt", "w");
    int counts[6] = {0}, value, invalid = 0, total = 0, sum = 0, star, mark, status;
    int ch;
    fprintf(file, "5\n4\n5\n3\nabc\n5\n2\n4\n7\n5\n1\n4\n0\n5\n4\n");
    fclose(file);
    file = fopen("ratings.txt", "r");
    while ((status = fscanf(file, "%d", &value)) != EOF) {
        if (status == 1) {
            if (value >= 1 && value <= 5) { counts[value]++; total++; sum += value; }
            else invalid++;
        } else {
            fscanf(file, "%*s");                     // نتخطى كلمة غير رقمية
            invalid++;
        }
    }
    fclose(file);
    file = fopen("report.txt", "w");
    for (star = 5; star >= 1; star--) {
        fprintf(file, "%d | ", star);
        for (mark = 0; mark < counts[star]; mark++) fputc('*', file);
        fprintf(file, " %d\n", counts[star]);
    }
    fprintf(file, "valid %d, invalid %d, mean %.2f\n", total, invalid, (double)sum / total);
    fclose(file);
    file = fopen("report.txt", "r");
    while ((ch = fgetc(file)) != EOF) putchar(ch);
    fclose(file);
    return 0;
}

#define MAX 50

double median(double values[], int size) {
    double copy[MAX], key;
    int idx, back;
    for (idx = 0; idx < size; idx++) copy[idx] = values[idx];   // لا نغير الأصل

#include <stdio.h>

double mean(double values[], int size) {
    double total = 0;
    int idx;
    for (idx = 0; idx < size; idx++) total += values[idx];
    return total / size;
}

double mySqrt(double value) {
    double guess = value > 1 ? value : 1;
    int iter;
    if (value <= 0) return 0;
    for (iter = 0; iter < 60; iter++)
        guess = (guess + value / guess) / 2;      // نيوتن
    return guess;
}

double correlation(double first[], double second[], int size) {
    double meanF = mean(first, size), meanS = mean(second, size);
    double cov = 0, varF = 0, varS = 0, devF, devS;
    int idx;
    for (idx = 0; idx < size; idx++) {
        devF = first[idx] - meanF;
        devS = second[idx] - meanS;
        cov += devF * devS;
        varF += devF * devF;
        varS += devS * devS;
    }
    return cov / mySqrt(varF * varS);
}

    printf("r(ads, sales)   = %.4f\n", correlation(ads, sales, 8));
    printf("r(ads, returns) = %.4f\n", correlation(ads, returns, 8));

title: دالة المتوسط المتحرك للمبيعات
level: medium

@ statement
اكتب دالة `movingAverage(values, size, window, result)` تحسب **المتوسط المتحرك** على نافذة من `window` يومًا لتنعيم المبيعات اليومية، وتعيد عدد القيم المحسوبة. قارن بين نافذة 3 ونافذة 5.
@ steps
- أول متوسط هو متوسط أول `window` قيمة.
- كل متوسط تالٍ: نضيف القيمة الجديدة ونطرح القيمة الخارجة من النافذة (مجموع منزلق).
- عدد النتائج `size - window + 1`.
@ flow
do: sum = مجموع أول window قيمة
do: result[0] = sum / window
for: idx = window ; idx < size ; idx++
  do: sum = sum + values[idx] - values[idx - window]
  do: result[idx - window + 1] = sum / window
@ code
#include <stdio.h>

int movingAverage(double values[], int size, int window, double result[]) {
    double sum = 0;
    int idx;
    for (idx = 0; idx < window; idx++) sum += values[idx];
    result[0] = sum / window;
    for (idx = window; idx < size; idx++) {
        sum += values[idx] - values[idx - window];   // النافذة تنزلق بخانة
        result[idx - window + 1] = sum / window;
    }
    return size - window + 1;
}

void printSeries(char label[], double values[], int size) {
    int idx;
    printf("%s", label);
    for (idx = 0; idx < size; idx++) printf(" %5.1f", values[idx]);
    printf("\n");
}

int main(void) {
    double sales[10] = {50, 72, 41, 66, 90, 58, 63, 95, 70, 81};
    double smooth[10];
    int count;
    printSeries("raw  :", sales, 10);
    count = movingAverage(sales, 10, 3, smooth);
    printSeries("MA(3):", smooth, count);
    count = movingAverage(sales, 10, 5, smooth);
    printSeries("MA(5):", smooth, count);
    return 0;
}

title: دالة عمر الزبون وشريحته
level: medium
domain: ds
@ statement
اكتب دالة `ageAt(birthDay, birthMonth, birthYear, day, month, year)` تحسب العمر بالسنوات الكاملة في تاريخ معين، ودالة `segment(age)` تعيد رقم الشريحة التسويقية (0: أقل من 18، 1: 18-24، 2: 25-39، 3: 40-59، 4: 60 فأكثر). احسب عدد الزبائن في كل شريحة.
@ steps
- العمر = فرق السنوات، ناقص 1 إذا لم يحل عيد الميلاد بعد في السنة الحالية.
- `segment` سلسلة من `if` تعيد رقمًا، و `main` تزيد عداد الشريحة.
@ flow
for: idx = 0 ; idx < 8 ; idx++
  call: age = ageAt(birth[idx], 6, 11, 2026)
  call: group = segment(age)
  do: counts[group]++
out: counts
@ code
#include <stdio.h>

int ageAt(int birthDay, int birthMonth, int birthYear, int day, int month, int year) {
    int age = year - birthYear;
    if (month < birthMonth || (month == birthMonth && day < birthDay))
        age--;                                   // عيد الميلاد لم يحل بعد
    return age;
}

int segment(int age) {
    if (age < 18) return 0;
    if (age < 25) return 1;
    if (age < 40) return 2;
    if (age < 60) return 3;
    return 4;
}

int main(void) {
    int birth[8][3] = {{12, 3, 2001}, {30, 11, 2008}, {6, 11, 1990}, {7, 11, 1967},
                       {1, 1, 1958}, {15, 8, 2004}, {20, 12, 1986}, {9, 2, 2010}};
    char labels[5][8] = {"<18", "18-24", "25-39", "40-59", "60+"};
    int counts[5] = {0}, idx, age;
    for (idx = 0; idx < 8; idx++) {
        age = ageAt(birth[idx][0], birth[idx][1], birth[idx][2], 6, 11, 2026);
        counts[segment(age)]++;
        printf("Customer %d: %2d years\n", idx + 1, age);
    }
    for (idx = 0; idx < 5; idx++)
        printf("%-6s %d\n", labels[idx], counts[idx]);
    return 0;
}

level: hard
domain: ds
@ statement
اكتب مجموعة دوال لتدريب نموذج **انحدار خطي** `sales = slope * ads + intercept` بطريقة المربعات الصغرى: `mean`، `fitSlope`، `fitIntercept`، `predict`، و `rSquared` (معامل التحديد). درّب النموذج ثم تنبأ بالمبيعات لميزانيات جديدة.
@ steps
- `slope = Σ(dx·dy) / Σ(dx²)` حيث `dx = x - meanX` و `dy = y - meanY`.
- `intercept = meanY - slope * meanX`.
- `R² = 1 - SSres / SStot`، حيث `SSres = Σ(y - predict(x))²` و `SStot = Σ(y - meanY)²`.
@ flow
call: slope = fitSlope(ads, sales, 8)
call: intercept = fitIntercept(ads, sales, 8, slope)
call: r2 = rSquared(ads, sales, 8, slope, intercept)
out: slope, intercept, r2
for: idx = 0 ; idx < 3 ; idx++
  call: predict(budgets[idx], slope, intercept)
@ code
#include <stdio.h>

double mean(double values[], int size) {
    double total = 0;
    int idx;
    for (idx = 0; idx < size; idx++) total += values[idx];
    return total / size;
}

double fitSlope(double xs[], double ys[], int size) {
    double meanX = mean(xs, size), meanY = mean(ys, size), num = 0, den = 0;
    int idx;
    for (idx = 0; idx < size; idx++) {
        num += (xs[idx] - meanX) * (ys[idx] - meanY);
        den += (xs[idx] - meanX) * (xs[idx] - meanX);
    }
    return num / den;
}

double fitIntercept(double xs[], double ys[], int size, double slope) {
    return mean(ys, size) - slope * mean(xs, size);   // المستقيم يمر بالنقطة المتوسطة
}

double predict(double value, double slope, double intercept) {
    return slope * value + intercept;
}

double rSquared(double xs[], double ys[], int size, double slope, double intercept) {
    double meanY = mean(ys, size), ssRes = 0, ssTot = 0, err;
    int idx;
    for (idx = 0; idx < size; idx++) {
        err = ys[idx] - predict(xs[idx], slope, intercept);
        ssRes += err * err;                               // خطأ النموذج
        ssTot += (ys[idx] - meanY) * (ys[idx] - meanY);   // التشتت الكلي
    }
    return 1 - ssRes / ssTot;
}

int main(void) {
    double ads[8] = {10, 15, 8, 20, 25, 12, 30, 18};
    double sales[8] = {210, 260, 190, 330, 370, 240, 420, 300};
    double budgets[3] = {5, 22, 40};
    double slope = fitSlope(ads, sales, 8);
    double intercept = fitIntercept(ads, sales, 8, slope);
    int idx;
    printf("sales = %.3f * ads + %.3f\n", slope, intercept);
    printf("R^2 = %.4f\n", rSquared(ads, sales, 8, slope, intercept));
    for (idx = 0; idx < 3; idx++)
        printf("Budget %4.0f -> predicted sales %.1f\n", budgets[idx],
               predict(budgets[idx], slope, intercept));
    return 0;
}

#include <stdio.h>

#define MAX 32

void mergeRuns(double source[], double target[], int left, int middle, int right) {
    int first = left, second = middle, out = left;
    while (first < middle && second < right)          // نأخذ الأصغر من الجزأين
        target[out++] = source[first] <= source[second] ? source[first++] : source[second++];
    while (first < middle) target[out++] = source[first++];
    while (second < right) target[out++] = source[second++];
}

void printArray(double values[], int size) {
    int idx;
    for (idx = 0; idx < size; idx++) printf("%.0f ", values[idx]);
    printf("\n");
}

void mergeSort(double values[], int size) {
    double buffer[MAX];
    int width, left, middle, right, idx;
    for (width = 1; width < size; width *= 2) {
        for (left = 0; left < size; left += 2 * width) {
            middle = left + width < size ? left + width : size;
            right = left + 2 * width < size ? left + 2 * width : size;
            mergeRuns(values, buffer, left, middle, right);
        }
        for (idx = 0; idx < size; idx++) values[idx] = buffer[idx];
        printf("width %2d: ", width);
        printArray(values, size);
    }
}

@ statement

@ steps
- المراكز الابتدائية: أصغر قيمة، قيمة وسطى، أكبر قيمة.
- `assign` تعيد عدد الزبائن الذين تغيرت مجموعتهم؛ نتوقف عندما يصبح 0.
- `updateCenters` تحسب متوسط كل مجموعة.
@ flow
do: centers = {min, وسط, max} ; changed = 1
while: changed > 0
  call: changed = assign(spend, groups, centers)
  call: updateCenters(spend, groups, centers)
  out: iteration, centers
out: groups
@ code
#include <stdio.h>

#define NB 12
#define NBK 3

int nearestCenter(double value, double centers[]) {
    int best = 0, idx;
    double dist, bestDist = -1;
    for (idx = 0; idx < NBK; idx++) {
        dist = value > centers[idx] ? value - centers[idx] : centers[idx] - value;
        if (bestDist < 0 || dist < bestDist) { bestDist = dist; best = idx; }
    }
    return best;
}

int assign(double spend[], int groups[], double centers[]) {
    int idx, group, changed = 0;
    for (idx = 0; idx < NB; idx++) {
        group = nearestCenter(spend[idx], centers);
        if (group != groups[idx]) { groups[idx] = group; changed++; }
    }
    return changed;
}

void updateCenters(double spend[], int groups[], double centers[]) {
    double sums[NBK] = {0};
    int counts[NBK] = {0}, idx;
    for (idx = 0; idx < NB; idx++) {
        sums[groups[idx]] += spend[idx];
        counts[groups[idx]]++;
    }
    for (idx = 0; idx < NBK; idx++)
        if (counts[idx] > 0) centers[idx] = sums[idx] / counts[idx];   // المركز = المتوسط
}

int main(void) {
    double spend[NB] = {12, 15, 9, 48, 52, 45, 110, 18, 55, 98, 120, 50};   // بالآلاف
    double centers[NBK] = {9, 50, 120};
    int groups[NB], idx, iteration = 0, changed = 1, group;
    for (idx = 0; idx < NB; idx++) groups[idx] = -1;
    while (changed > 0) {
        changed = assign(spend, groups, centers);
        updateCenters(spend, groups, centers);
        iteration++;
        printf("Iteration %d: %d changed, centers %.1f %.1f %.1f\n",
               iteration, changed, centers[0], centers[1], centers[2]);
    }
    for (group = 0; group < NBK; group++) {
        printf("Group %d:", group);
        for (idx = 0; idx < NB; idx++)
            if (groups[idx] == group) printf(" %.0f", spend[idx]);
        printf("\n");
    }
    return 0;
}

@ steps
- `best[item][money]` = أكبر ربح باستعمال أول `item` منتجًا وميزانية `money`.
- إما لا نأخذ المنتج: `best[item-1][money]`، أو نأخذه: `gains + best[item-1][money - cost]`.
- إعادة البناء: إذا اختلف `best[item][money]` عن `best[item-1][money]` فالمنتج مأخوذ.
@ flow

  for: money = 0 ; money <= budget ; money++
    do: best[item][money] = best[item-1][money]
    if: costs[item-1] <= money && أخذه أفضل
      do: best[item][money] = gains[item-1] + best[item-1][money - costs[item-1]]
do: إعادة البناء من best[size][budget]
out: best[size][budget], chosen
@ code
#include <stdio.h>

#define NB 6
#define BUDGET 50

int knapsack(int costs[], int gains[], int size, int budget, int chosen[]) {
    int best[NB + 1][BUDGET + 1], item, money, withItem;
    for (money = 0; money <= budget; money++) best[0][money] = 0;
    for (item = 1; item <= size; item++)
        for (money = 0; money <= budget; money++) {
            best[item][money] = best[item - 1][money];             // لا نأخذه
            if (costs[item - 1] <= money) {
                withItem = gains[item - 1] + best[item - 1][money - costs[item - 1]];
                if (withItem > best[item][money]) best[item][money] = withItem;
            }
        }
    money = budget;
    for (item = size; item >= 1; item--) {                         // إعادة البناء
        chosen[item - 1] = best[item][money] != best[item - 1][money];
        if (chosen[item - 1]) money -= costs[item - 1];
    }
    return best[size][budget];
}

int main(void) {
    char names[NB][12] = {"Phone", "Laptop", "Earbuds", "Watch", "Tablet", "Speaker"};
    int costs[NB] = {20, 30, 10, 15, 25, 5};       // بالآلاف
    int gains[NB] = {60, 90, 35, 40, 70, 18};
    int chosen[NB], idx, spent = 0;
    int total = knapsack(costs, gains, NB, BUDGET, chosen);
    for (idx = 0; idx < NB; idx++)
        if (chosen[idx]) {
            printf("Promote %-8s cost %2d gain %2d\n", names[idx], costs[idx], gains[idx]);
            spent += costs[idx];
        }
    printf("Budget used %d/%d, expected gain %d (thousand DA)\n", spent, BUDGET, total);
    return 0;
}

#define NS 3

void multiplyVector(double vec[], double matrix[NS][NS], double result[]) {
    int from, to;
    for (to = 0; to < NS; to++) {
        result[to] = 0;
        for (from = 0; from < NS; from++)
            result[to] += vec[from] * matrix[from][to];   // من كل حالة نحو to
    }
}

void multiplyMatrix(double left[NS][NS], double right[NS][NS], double result[NS][NS]) {
    int row, col, mid;
    for (row = 0; row < NS; row++)
        for (col = 0; col < NS; col++) {
            result[row][col] = 0;
            for (mid = 0; mid < NS; mid++)
                result[row][col] += left[row][mid] * right[mid][col];
        }
}

@ steps

@ flow

  do: middle = (low + high) / 2
  if: profit(low) * profit(middle) <= 0

#include <stdio.h>

double demand(double price) {
    return 120 / (1 + 0.04 * price * price);
}

double profit(double price) {
    return (price - 2) * demand(price) - 40;
}

double bisection(double low, double high, double tolerance) {
    double middle;
    int steps = 0;
    while (high - low > tolerance) {
        middle = (low + high) / 2;
        if (profit(low) * profit(middle) <= 0) high = middle;   // الجذر في النصف الأيسر
        else low = middle;
        steps++;
    }
    printf("Bisection: %d steps\n", steps);
    return (low + high) / 2;
}

double bestPrice(double low, double high) {
    double third1, third2;
    while (high - low > 1e-6) {
        third1 = low + (high - low) / 3;
        third2 = high - (high - low) / 3;
        if (profit(third1) < profit(third2)) low = third1;    // القمة ليست في الثلث الأيسر
        else high = third2;
    }
    return (low + high) / 2;
}

#include <stdio.h>

double rate(double hour) {
    return 20 + 60 * hour * (24 - hour) / 144;
}

double absValue(double value) { return value < 0 ? -value : value; }

double trapezoid(double start, double end, int steps) {
    double width = (end - start) / steps, sum = (rate(start) + rate(end)) / 2;
    int idx;
    for (idx = 1; idx < steps; idx++) sum += rate(start + idx * width);
    return sum * width;
}

double simpson(double start, double end, int steps) {
    double width = (end - start) / steps, sum = rate(start) + rate(end);
    int idx;
    for (idx = 1; idx < steps; idx++)
        sum += (idx % 2 == 1 ? 4 : 2) * rate(start + idx * width);   // أوزان 4 و 2
    return sum * width / 3;
}

- نبدأ باعتبار كل الأعداد أولية؛ لكل أولي `prime` نشطب مضاعفاته بدءًا من `prime * prime`.
- التحليل: نقسم العدد على كل أولي ما دام القسمة ممكنة ونعد الأس.

#include <stdio.h>

#define LIMIT 10000

void sieve(int limit, char isPrime[]) {
    int number, multiple;
    for (number = 0; number <= limit; number++) isPrime[number] = number >= 2;
    for (number = 2; number * number <= limit; number++)
        if (isPrime[number])
            for (multiple = number * number; multiple <= limit; multiple += number)
                isPrime[multiple] = 0;                // مضاعف = غير أولي
}

void factorize(int number, char isPrime[]) {
    int prime, exponent, first = 1;
    printf("%d =", number);
    for (prime = 2; prime <= number; prime++) {
        if (!isPrime[prime]) continue;
        exponent = 0;
        while (number % prime == 0) {
            number /= prime;
            exponent++;
        }
        if (exponent > 0) {
            printf("%s %d", first ? "" : " x", prime);
            if (exponent > 1) printf("^%d", exponent);
            first = 0;
        }
    }
    printf("\n");
}

int main(void) {
    char isPrime[LIMIT + 1];
    int number, query, count = 0;
    sieve(LIMIT, isPrime);
    for (number = 0; number <= LIMIT; number++) count += isPrime[number];
    printf("%d primes up to %d\n", count, LIMIT);
    for (query = 0; query < 4; query++) {
        scanf("%d", &number);
        factorize(number, isPrime);
    }
    return 0;
}

- زيلر: نعتبر جانفي وفيفري الشهرين 13 و 14 من السنة السابقة، ثم `(day + 13(month+1)/5 + yearOfCentury + yearOfCentury/4 + century/4 + 5*century) % 7` (0 = السبت).
- `addBusinessDays`: نتقدم يومًا يومًا، ونعد اليوم فقط إذا لم يكن جمعة أو سبتًا.
- الفرق بين تاريخين = فرق `dayNumber`.
@ flow

  call: dayOfWeek(تاريخ الطلبية)
  call: addBusinessDays(التاريخ, delays[order])
  out: تاريخ التسليم واليوم
call: dayNumber(تاريخ1) - dayNumber(تاريخ2)
@ code
#include <stdio.h>

int daysInMonth(int month, int year) {
    int days[12] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    if (month == 2 && ((year % 4 == 0 && year % 100 != 0) || year % 400 == 0)) return 29;
    return days[month - 1];
}

int dayOfWeek(int day, int month, int year) {    // 0 = السبت، 6 = الجمعة
    int century, yearOfCentury;
    if (month < 3) { month += 12; year--; }
    century = year / 100;
    yearOfCentury = year % 100;
    return (day + 13 * (month + 1) / 5 + yearOfCentury + yearOfCentury / 4
            + century / 4 + 5 * century) % 7;
}

long dayNumber(int day, int month, int year) {
    long total = 0;
    int walk;
    for (walk = 2000; walk < year; walk++) total += 365 + (daysInMonth(2, walk) == 29);
    for (walk = 1; walk < month; walk++) total += daysInMonth(walk, year);
    return total + day - 1;
}

void addBusinessDays(int date[], int count) {
    int weekday;
    while (count > 0) {
        date[0]++;                                   // اليوم التالي
        if (date[0] > daysInMonth(date[1], date[2])) {
            date[0] = 1;
            if (++date[1] > 12) { date[1] = 1; date[2]++; }
        }
        weekday = dayOfWeek(date[0], date[1], date[2]);
        if (weekday != 6 && weekday != 0) count--;   // نتخطى الجمعة والسبت
    }
}

int main(void) {
    char names[7][4] = {"Sat", "Sun", "Mon", "Tue", "Wed", "Thu", "Fri"};
    int orders[3][3] = {{27, 11, 2026}, {29, 12, 2026}, {25, 2, 2028}};
    int delays[3] = {3, 5, 4}, date[3], order;
    for (order = 0; order < 3; order++) {
        date[0] = orders[order][0]; date[1] = orders[order][1]; date[2] = orders[order][2];
        printf("%s %02d/%02d/%d + %d business days -> ", names[dayOfWeek(date[0], date[1], date[2])],
               date[0], date[1], date[2], delays[order]);
        addBusinessDays(date, delays[order]);
        printf("%s %02d/%02d/%d\n", names[dayOfWeek(date[0], date[1], date[2])],
               date[0], date[1], date[2]);
    }
    printf("Days between 01/03/2026 and 06/11/2026: %ld\n",
           dayNumber(6, 11, 2026) - dayNumber(1, 3, 2026));
    return 0;
}

@ steps
- المسافة الإقليدية المربعة كافية للمقارنة (لا حاجة للجذر).
- `findNearest` تختار `k` مرات أقرب نقطة لم تُختر بعد.
- التصنيف = الفئة الأكثر بين الجيران `k`.
@ flow

  call: findNearest(newVisits[query], newBasket[query], nearest)

  if: votes * 2 > NBK
    out: `"buyer"`

@ code
#include <stdio.h>

#define NB 10
#define NBK 3

double distance2(double visitsA, double basketA, double visitsB, double basketB) {
    return (visitsA - visitsB) * (visitsA - visitsB) + (basketA - basketB) * (basketA - basketB);
}

void findNearest(double visits[], double basket[], double qVisits, double qBasket, int nearest[]) {
    int used[NB] = {0}, rank, idx, best;
    double dist, bestDist;
    for (rank = 0; rank < NBK; rank++) {
        best = -1;
        bestDist = 0;
        for (idx = 0; idx < NB; idx++) {
            if (used[idx]) continue;
            dist = distance2(visits[idx], basket[idx], qVisits, qBasket);
            if (best < 0 || dist < bestDist) { best = idx; bestDist = dist; }
        }
        used[best] = 1;                      // لا نختار نفس الجار مرتين
        nearest[rank] = best;
    }
}

int classify(double visits[], double basket[], int labels[], double qVisits, double qBasket) {
    int nearest[NBK], rank, votes = 0;
    findNearest(visits, basket, qVisits, qBasket, nearest);
    printf("  neighbours:");
    for (rank = 0; rank < NBK; rank++) {
        printf(" #%d", nearest[rank]);
        votes += labels[nearest[rank]];
    }
    return votes * 2 > NBK;                    // أغلبية الأصوات
}

int main(void) {
    double visits[NB] = {1, 2, 8, 9, 3, 10, 2, 7, 6, 1};
    double basket[NB] = {2.0, 1.5, 6.0, 7.5, 2.5, 5.0, 3.0, 8.0, 4.5, 1.0};
    int labels[NB] = {0, 0, 1, 1, 0, 1, 0, 1, 1, 0};
    double newVisits[3] = {7, 2, 5}, newBasket[3] = {5.5, 2.0, 3.5};
    int query, result;
    for (query = 0; query < 3; query++) {
        printf("Customer (%.0f visits, %.1fk):", newVisits[query], newBasket[query]);
        result = classify(visits, basket, labels, newVisits[query], newBasket[query]);
        printf(" -> %s\n", result ? "buyer" : "not buyer");
    }
    return 0;
}

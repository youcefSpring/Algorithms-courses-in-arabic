#include <stdio.h>

#define MAXLEN 40

int main(void) {
    char first[] = "fast usb charger 20w";
    char second[] = "usb-c fast charger";
    int lcs[MAXLEN][MAXLEN];
    char common[MAXLEN];
    int lenA = 0, lenB = 0, row, col, size;
    while (first[lenA] != '\0') lenA++;
    while (second[lenB] != '\0') lenB++;
    for (row = 0; row <= lenA; row++)
        for (col = 0; col <= lenB; col++) {
            if (row == 0 || col == 0)
                lcs[row][col] = 0;                         // نص فارغ
            else if (first[row - 1] == second[col - 1])
                lcs[row][col] = lcs[row - 1][col - 1] + 1;
            else
                lcs[row][col] = lcs[row - 1][col] > lcs[row][col - 1]
                                ? lcs[row - 1][col] : lcs[row][col - 1];
        }
    size = lcs[lenA][lenB];
    common[size] = '\0';
    row = lenA; col = lenB;
    while (row > 0 && col > 0) {                           // إعادة البناء من النهاية
        if (first[row - 1] == second[col - 1]) {
            common[--size] = first[row - 1];
            row--; col--;
        } else if (lcs[row - 1][col] >= lcs[row][col - 1]) {
            row--;
        } else {
            col--;
        }
    }
    printf("LCS length: %d\n", lcs[lenA][lenB]);
    printf("LCS: \"%s\"\n", common);
    printf("Similarity: %.2f\n", 2.0 * lcs[lenA][lenB] / (lenA + lenB));

    int counts[MAXW], nbWords = 0, len = 0, pos, idx, found, same, letter, tmpCount, back;

- الفراغات = `WIDTH - مجموع الأطوال`، تُقسم على عدد الفجوات، والباقي يوزع على الفجوات الأولى.
- السطر الأخير أو سطر بكلمة واحدة: محاذاة عادية إلى اليسار.
@ flow
do: استخراج الكلمات (start, len)
do: first = 0
while: first < nbWords
  do: last = first ; lineLen = len[first]
  while: last + 1 < nbWords && lineLen + 1 + len[last+1] <= WIDTH
    do: last++ ; lineLen += 1 + len[last]
  if: last == nbWords - 1 || last == first
    out: كلمات السطر بفراغ واحد
  else:
    do: gaps = last - first ; spaces = WIDTH - مجموع الأطوال
    out: كل فجوة: spaces / gaps (+1 للفجوات الأولى)
  do: first = last + 1
@ code
#include <stdio.h>

#define WIDTH 24

int main(void) {
    char text[] = "Wireless noise cancelling headphones with long battery life and fast charging case";
    int start[40], len[40], nbWords = 0, pos = 0;
    int first, last, lineLen, letters, gaps, spaces, word, gap, extra, col;
    while (text[pos] != '\0') {                         // استخراج الكلمات
        while (text[pos] == ' ') pos++;
        if (text[pos] == '\0') break;
        start[nbWords] = pos;
        while (text[pos] != ' ' && text[pos] != '\0') pos++;
        len[nbWords] = pos - start[nbWords];
        nbWords++;
    }
    printf("+------------------------+\n");
    for (first = 0; first < nbWords; first = last + 1) {
        last = first;
        lineLen = len[first];
        letters = len[first];
        while (last + 1 < nbWords && lineLen + 1 + len[last + 1] <= WIDTH) {
            last++;
            lineLen += 1 + len[last];
            letters += len[last];
        }
        gaps = last - first;
        spaces = WIDTH - letters;
        col = 0;
        putchar('|');
        for (word = first; word <= last; word++) {
            for (pos = 0; pos < len[word]; pos++) putchar(text[start[word] + pos]);
            col += len[word];
            if (word < last) {
                if (last == nbWords - 1) extra = 1;     // السطر الأخير: فراغ واحد
                else extra = spaces / gaps + (word - first < spaces % gaps ? 1 : 0);
                for (gap = 0; gap < extra; gap++) putchar(' ');
                col += extra;
            }
        }
        while (col++ < WIDTH) putchar(' ');             // إكمال السطر
        printf("|\n");
    }
    printf("+------------------------+\n");
    return 0;
}

@@
title: ضغط النص بالقاموس على طريقة LZW
level: hard
domain: ds
@ statement
لتقليل حجم سجلات النقرات، اكتب برنامجًا يضغط سلسلة من الحروف `A..Z` بخوارزمية **LZW**: القاموس يبدأ بالحروف المفردة (الرموز 0..25)، ثم يضيف كل تسلسل جديد يصادفه. اعرض الرموز الناتجة ونسبة الضغط.
@ steps
- القاموس: جدول سلاسل `dict[code]`، والرموز 0..25 للحروف.
- `current` = أطول تسلسل موجود في القاموس؛ نجرب `current + next`.
- إن وُجد في القاموس نمدد `current`، وإلا نُخرج رمز `current`، ونضيف `current + next`، ونبدأ من `next`.
- نسبة الضغط: (عدد الرموز × 9 بتات) / (عدد الحروف × 8 بتات).
@ flow
do: dict[0..25] = A..Z ; current = text[0]
for: pos = 1 ; pos <= length ; pos++
  do: candidate = current + text[pos]
  if: pos < length && candidate في القاموس
    do: current = candidate
  else:
    out: رمز current
    do: dict[size++] = candidate
    do: current = text[pos]
out: نسبة الضغط
@ code
#include <stdio.h>

#define MAXD 128
#define MAXL 16

int main(void) {
    char text[] = "ABABABABCABCABCABCABAB";
    char dict[MAXD][MAXL], candidate[MAXL];
    int size = 26, length = 0, pos, code, letter, curLen, found, same, outCodes = 0, currentCode;
    for (code = 0; code < 26; code++) {               // القاموس الابتدائي
        dict[code][0] = (char)('A' + code);
        dict[code][1] = '\0';
    }
    while (text[length] != '\0') length++;
    candidate[0] = text[0];
    curLen = 1;
    currentCode = text[0] - 'A';
    printf("Codes:");
    for (pos = 1; pos <= length; pos++) {
        candidate[curLen] = text[pos];                // current + next
        candidate[curLen + 1] = '\0';
        found = -1;
        for (code = 0; pos < length && code < size && found < 0; code++) {
            same = 1;
            for (letter = 0; letter <= curLen + 1; letter++)
                if (dict[code][letter] != candidate[letter]) { same = 0; break; }
            if (same) found = code;
        }
        if (found >= 0) {
            currentCode = found;                      // نمدد التسلسل
            curLen++;
        } else {
            printf(" %d", currentCode);
            outCodes++;
            if (size < MAXD && curLen + 1 < MAXL) {   // تسلسل جديد في القاموس
                for (letter = 0; letter <= curLen + 1; letter++) dict[size][letter] = candidate[letter];
                size++;
            }
            candidate[0] = text[pos];
            curLen = 1;
            currentCode = text[pos] - 'A';
        }
    }
    printf("\n%d chars -> %d codes, dictionary size %d\n", length, outCodes, size);
    printf("Compression ratio: %.2f\n", (outCodes * 9.0) / (length * 8.0));
    return 0;
}

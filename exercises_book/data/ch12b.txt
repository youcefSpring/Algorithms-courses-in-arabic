#include <stdlib.h>

typedef struct Digit { int value; struct Digit *next; } Digit;

Digit *fromText(const char *text) {
    Digit *head = NULL, *node;
    for (; *text != '\0'; text++) {
        node = malloc(sizeof(Digit));
        node->value = *text - '0';
        node->next = head;                    // الآحاد تنتهي في الرأس
        head = node;
    }
    return head;
}

Digit *addNumbers(const Digit *first, const Digit *second) {
    Digit dummy, *tail = &dummy, *node;
    int carry = 0, digit;
    while (first != NULL || second != NULL || carry) {
        digit = carry + (first ? first->value : 0) + (second ? second->value : 0);
        node = malloc(sizeof(Digit));
        node->value = digit % 10;
        node->next = NULL;
        tail->next = node;
        tail = node;
        carry = digit / 10;                   // الحمل
        if (first) first = first->next;
        if (second) second = second->next;
    }
    return dummy.next;
}

void printNumber(const Digit *head) {
    if (head == NULL) return;
    printNumber(head->next);                  // الأرقام الأعلى أولًا
    printf("%d", head->value);
}

void freeDigits(Digit *head) { Digit *next; for (; head != NULL; head = next) { next = head->next; free(head); } }

    printf("\n= "); printNumber(sum); printf("\n");

#include <stdlib.h>

Node *fromArray(const int *values, int size) {
    Node *head = NULL, *node;
    while (size-- > 0) { node = malloc(sizeof(Node)); node->id = values[size]; node->next = head; head = node; }
    return head;
}

int length(const Node *head) { int count = 0; for (; head != NULL; head = head->next) count++; return count; }
void freeList(Node *head) { Node *next; for (; head != NULL; head = next) { next = head->next; free(head); } }

Node *intersection(const Node *first, const Node *second) {
    Node dummy, *tail = &dummy, *node;
    dummy.next = NULL;
    while (first != NULL && second != NULL) {
        if (first->id == second->id) {
            node = malloc(sizeof(Node));
            node->id = first->id;
            node->next = NULL;
            tail->next = node;
            tail = node;
            first = first->next;
            second = second->next;

            first = first->next;                 // الأصغر يتقدم
        } else {
            second = second->next;
        }
    }
    return dummy.next;
}

Entry *build(const int *indexes, const double *values, int size) {

    while (size-- > 0) {
        node = malloc(sizeof(Entry));

double dot(const Entry *first, const Entry *second) {
    double sum = 0;
    while (first != NULL && second != NULL) {
        if (first->index == second->index) {
            sum += first->value * second->value;
            first = first->next; second = second->next;
        } else if (first->index < second->index) {
            first = first->next;
        } else {
            second = second->next;

#include <stdlib.h>

Node *rotate(Node *head, int shift) {
    Node *tail = head, *walk;
    int length = 1, step;
    if (head == NULL) return head;
    while (tail->next != NULL) { tail = tail->next; length++; }

    if (shift == 0) return head;

#include <stdlib.h>

typedef struct Node { char ch; struct Node *next; } Node;

Node *reverse(Node *head) {
    Node *prev = NULL, *next;
    while (head != NULL) { next = head->next; head->next = prev; prev = head; head = next; }
    return prev;
}

int isPalindrome(Node *head) {
    Node *slow = head, *fast = head, *second, *left, *right;

    reverse(second);                                     // نعيده كما كان
    return same;
}

@ steps
- `link` يشير إلى الحقل الذي يحمل العقدة الأولى من الزوج.
- `first = *link`, `second = first->next`: `first->next = second->next; second->next = first; *link = second;`.

@ flow

@ code
#include <stdio.h>
#include <stdlib.h>

typedef struct Node { int code; struct Node *next; } Node;

Node *swapPairs(Node *head) {
    Node **link = &head, *first, *second;
    while (*link != NULL && (*link)->next != NULL) {
        first = *link;
        second = first->next;
        first->next = second->next;         // إعادة الربط
        second->next = first;
        *link = second;
        link = &first->next;                // الزوج التالي
    }
    return head;
}

level: medium

@ statement
قائمة انتظار المنتجات النافدة **مزدوجة الربط** (رأس وذيل). اكتب `removeNode(list, node)` تحذف أي عقدة في وقت ثابت بتحديث جاريها (دون البحث عن السابقة)، و `printForward` و `printBackward` للتحقق من الاتجاهين بعد كل حذف.
@ steps
- إذا كان للعقدة سابق: `node->prev->next = node->next`، وإلا `list->head = node->next`.
- إذا كان لها تالٍ: `node->next->prev = node->prev`، وإلا `list->tail = node->prev`.
@ flow

call: removeNode(&list, الرأس)

@ code
#include <stdio.h>
#include <stdlib.h>

typedef struct Node { int code; struct Node *prev, *next; } Node;
typedef struct { Node *head, *tail; } List;

void pushBack(List *list, int code) {
    Node *node = malloc(sizeof(Node));
    node->code = code; node->next = NULL; node->prev = list->tail;
    if (list->tail != NULL) list->tail->next = node; else list->head = node;
    list->tail = node;
}

void removeNode(List *list, Node *node) {
    if (node->prev != NULL) node->prev->next = node->next; else list->head = node->next;
    if (node->next != NULL) node->next->prev = node->prev; else list->tail = node->prev;
    free(node);                                         // دون بحث: وقت ثابت
}

void printBoth(const char *label, const List *list) {
    const Node *walk;
    printf("%-14s fwd:", label);
    for (walk = list->head; walk != NULL; walk = walk->next) printf(" %d", walk->code);
    printf("  bwd:");
    for (walk = list->tail; walk != NULL; walk = walk->prev) printf(" %d", walk->code);
    printf("\n");
}

int main(void) {
    List list = {NULL, NULL};
    int idx;
    for (idx = 0; idx < 6; idx++) pushBack(&list, 101 + idx);
    printBoth("start", &list);
    removeNode(&list, list.head->next->next);  printBoth("remove 103", &list);
    removeNode(&list, list.head);              printBoth("remove head", &list);
    removeNode(&list, list.tail);              printBoth("remove tail", &list);
    while (list.head != NULL) removeNode(&list, list.head);
    return 0;
}

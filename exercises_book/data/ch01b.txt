    dq = (Q2 - Q1) / ((Q1 + Q2) / 2);     // التغير النسبي للكمية
    dp = (P2 - P1) / ((P1 + P2) / 2);     // التغير النسبي للسعر
    elasticity = dq / dp;                          // المرونة
    printf("Elasticity = %.3f\n", elasticity);
    printf("Revenue before = %.0f DA\n", P1 * Q1);
    printf("Revenue after  = %.0f DA\n", P2 * Q2);
    return 0;
}

title: تعبئة حاوية الشحن
level: hard
domain: ec
@ statement
حاوية شحن أبعادها `contLength × contWidth × contHeight` (سم) ونريد ملأها بعلب أبعادها `boxLength × boxWidth × boxHeight` مرتبة في نفس الاتجاه. اكتب برنامجًا يحسب عدد العلب في كل بعد، والعدد الكلي، و**نسبة الحجم المستغل**.
@ stdin
600 240 250
40 30 50
@ steps
- عدد العلب في كل بعد: `contLength / boxLength` و`contWidth / boxWidth` و`contHeight / boxHeight` (قسمة صحيحة).
- العدد الكلي: جداء الأعداد الثلاثة.
- الحجم المستغل: `nbBoxes * boxLength * boxWidth * boxHeight`، والنسبة بالمئة من `contLength * contWidth * contHeight`.
@ flow
in: contLength, contWidth, contHeight, boxLength, boxWidth, boxHeight
do: nx = contLength / boxLength ; ny = contWidth / boxWidth ; nz = contHeight / boxHeight
do: nbBoxes = nx * ny * nz
do: used = nbBoxes * boxLength * boxWidth * boxHeight
do: pct = used / (contLength * contWidth * contHeight) * 100
out: nx, ny, nz, nbBoxes, pct
@ code
#include <stdio.h>

int main(void) {
    long contLength, contWidth, contHeight, boxLength, boxWidth, boxHeight, nx, ny, nz, nbBoxes, used;
    double pct;
    scanf("%ld %ld %ld", &contLength, &contWidth, &contHeight);     // أبعاد الحاوية
    scanf("%ld %ld %ld", &boxLength, &boxWidth, &boxHeight);     // أبعاد العلبة
    nx = contLength / boxLength;                   // علب على الطول
    ny = contWidth / boxWidth;                   // علب على العرض
    nz = contHeight / boxHeight;                   // طبقات على الارتفاع
    nbBoxes = nx * ny * nz;             // العدد الكلي
    used = nbBoxes * boxLength * boxWidth * boxHeight;         // الحجم المشغول بالعلب
    pct = (double)used / (contLength * contWidth * contHeight) * 100;
    printf("Grid: %ld x %ld x %ld\n", nx, ny, nz);
    printf("Boxes = %ld\n", nbBoxes);
    printf("Volume used = %.2f %%\n", pct);
    return 0;
}

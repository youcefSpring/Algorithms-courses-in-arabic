@ steps
- نبدأ بـ `guess = value` (أو 1 إذا كان `value < 1`)، ونعيد 0 إذا كان `value == 0`.

#include <stdio.h>

double mySqrt(double value) {
    double guess, previous;
    if (value <= 0) return 0;
    guess = value > 1 ? value : 1;
    do {
        previous = guess;
        guess = (guess + value / guess) / 2;      // صيغة نيوتن
    } while (previous - guess > 1e-9 || guess - previous > 1e-9);
    return guess;
}

int main(void) {
    double sales[7] = {52, 61, 48, 70, 66, 59, 64};
    double mean = 0, variance = 0;
    int day;
    for (day = 0; day < 7; day++) mean += sales[day];
    mean /= 7;
    for (day = 0; day < 7; day++)
        variance += (sales[day] - mean) * (sales[day] - mean);
    variance /= 7;
    printf("sqrt(2) = %.9f\n", mySqrt(2));
    printf("Mean = %.2f, std = %.3f\n", mean, mySqrt(variance));
    return 0;
}

#include <stdio.h>

int gcd(int first, int second) {
    int rest;
    while (second != 0) {
        rest = first % second;          // إقليدس
        first = second;
        second = rest;
    }
    return first;
}

int main(void) {
    int pens = 84, notebooks = 120, erasers = 156;
    int boxes = gcd(gcd(pens, notebooks), erasers);   // استدعاء متداخل

#include <stdio.h>

int isPrime(int number) {
    int divisor;
    if (number < 2) return 0;
    if (number % 2 == 0) return number == 2;
    for (divisor = 3; divisor * divisor <= number; divisor += 2)   // حتى الجذر فقط
        if (number % divisor == 0)
            return 0;
    return 1;
}

        if (isPrime(code)) {
            printf("%d ", code);
            count++;
        }
    printf("\n%d valid coupons\n", count);

@ stdin
4539148803436467 4539148803436468
@ steps
- نمر على الأرقام من اليمين إلى اليسار؛ نضاعف رقمًا من كل اثنين (الثاني، الرابع، ...).
- إذا تجاوز الرقم المضاعف 9 نطرح منه 9.
- الرقم صالح إذا كان المجموع يقبل القسمة على 10.
@ flow
in: card
do: sum = 0 ; double = 0
for: pos = length - 1 ; pos >= 0 ; pos--
  do: digit = card[pos] - '0'
  if: double
    do: digit = digit * 2 ; إذا digit > 9 نطرح 9
  do: sum = sum + digit ; double = !double
out: sum % 10 == 0
@ code
#include <stdio.h>

int luhnValid(char number[]) {
    int length = 0, pos, digit, sum = 0, doubled = 0;
    while (number[length] != '\0') length++;
    for (pos = length - 1; pos >= 0; pos--) {     // من اليمين
        digit = number[pos] - '0';
        if (doubled) {
            digit *= 2;
            if (digit > 9) digit -= 9;
        }
        sum += digit;
        doubled = !doubled;                       // رقم من كل اثنين
    }
    return sum % 10 == 0;
}

int main(void) {
    char card[25];
    int idx;
    for (idx = 0; idx < 2; idx++) {
        scanf("%24s", card);
        printf("%s: %s\n", card, luhnValid(card) ? "valid" : "invalid");
    }
    return 0;
}

#include <stdio.h>

void minMaxNormalize(double values[], double normalized[], int size) {
    double low = values[0], high = values[0];
    int idx;
    for (idx = 1; idx < size; idx++) {
        if (values[idx] < low) low = values[idx];
        if (values[idx] > high) high = values[idx];
    }
    for (idx = 0; idx < size; idx++)
        normalized[idx] = high == low ? 0 : (values[idx] - low) / (high - low);
}

    int idx;
    printf("%s", label);
    for (idx = 0; idx < size; idx++) printf(" %.3f", values[idx]);
    printf("\n");
}

    printRow("prices:", result, 5);
    minMaxNormalize(agesD, result, 5);           // نفس الدالة لعمود آخر
    printRow("ages:  ", result, 5);

double myExp(double value) {

        sum += term;
        rank++;
    }
    return sum;
}

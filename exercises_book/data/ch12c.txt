#include <stdio.h>
#include <stdlib.h>

typedef struct Node { int number; double amount; struct Node *next; } Node;

Node *merge(Node *first, Node *second) {
    Node dummy, *tail = &dummy;
    while (first != NULL && second != NULL) {
        if (first->amount <= second->amount) { tail->next = first; first = first->next; }
        else { tail->next = second; second = second->next; }
        tail = tail->next;
    }
    tail->next = first != NULL ? first : second;
    return dummy.next;
}

Node *mergeSort(Node *head) {
    Node *slow, *fast, *second;
    if (head == NULL || head->next == NULL) return head;
    slow = head;
    fast = head->next;
    while (fast != NULL && fast->next != NULL) { slow = slow->next; fast = fast->next->next; }
    second = slow->next;                        // النصف الثاني
    slow->next = NULL;                          // نقطع القائمة
    return merge(mergeSort(head), mergeSort(second));
}

#include <stdio.h>
#include <stdlib.h>

#define CAP 3
#define MAXCODE 100

typedef struct Node { int code; struct Node *prev, *next; } Node;

Node *head = NULL, *tail = NULL, *slots[MAXCODE];
int size = 0;

void detach(Node *node) {
    if (node->prev) node->prev->next = node->next; else head = node->next;
    if (node->next) node->next->prev = node->prev; else tail = node->prev;
}

void pushFront(Node *node) {
    node->prev = NULL;
    node->next = head;
    if (head) head->prev = node; else tail = node;
    head = node;
}

int main(void) {
    Node *node, *walk;
    int code, hits = 0, misses = 0;
    while (scanf("%d", &code) == 1) {
        node = slots[code];
        if (node != NULL) {                            // إصابة
            hits++;
            detach(node);
            pushFront(node);
            printf("get %d: HIT  |", code);
        } else {                                       // إخفاق
            misses++;
            node = malloc(sizeof(Node));
            node->code = code;
            pushFront(node);
            slots[code] = node;

                detach(walk);

                printf("get %d: MISS |", code);

        for (walk = head; walk != NULL; walk = walk->next) printf(" %d", walk->code);
        printf("\n");

@ statement

@ steps
- `addTerm(&result, coef, power)`: نبحث عن موضع الأس؛ إن وُجد نجمع المعاملين (ونحذف العقدة إذا أصبح 0)، وإلا ندرج عقدة جديدة.

@ flow
for: termA = first ; termA != NULL ; termA = termA->next
  for: termB = second ; termB != NULL ; termB = termB->next

@ code
#include <stdio.h>
#include <stdlib.h>

typedef struct Term { double coef; int power; struct Term *next; } Term;

void addTerm(Term **poly, double coef, int power) {
    Term **link = poly, *node;
    while (*link != NULL && (*link)->power > power) link = &(*link)->next;
    if (*link != NULL && (*link)->power == power) {
        (*link)->coef += coef;                           // حد متشابه
        if ((*link)->coef == 0) { node = *link; *link = node->next; free(node); }
        return;
    }
    if (coef == 0) return;
    node = malloc(sizeof(Term));
    node->coef = coef;
    node->power = power;
    node->next = *link;                                  // إدراج مرتب
    *link = node;
}

Term *multiplyPoly(const Term *first, const Term *second) {
    Term *result = NULL;
    const Term *termA, *termB;
    for (termA = first; termA != NULL; termA = termA->next)
        for (termB = second; termB != NULL; termB = termB->next)
            addTerm(&result, termA->coef * termB->coef, termA->power + termB->power);
    return result;
}

void printPoly(const char *label, const Term *poly) {
    printf("%s", label);
    for (; poly != NULL; poly = poly->next) printf(" %+g t^%d", poly->coef, poly->power);
    printf("\n");
}

void freePoly(Term *poly) { Term *next; for (; poly != NULL; poly = next) { next = poly->next; free(poly); } }

int main(void) {
    Term *price = NULL, *quantity = NULL, *revenue;
    addTerm(&price, 3000, 0); addTerm(&price, -50, 1);                   // السعر ينخفض
    addTerm(&quantity, 200, 0); addTerm(&quantity, 15, 1); addTerm(&quantity, -0.5, 2);
    revenue = multiplyPoly(price, quantity);
    printPoly("price(t)   =", price);
    printPoly("quantity(t)=", quantity);
    printPoly("revenue(t) =", revenue);
    freePoly(price); freePoly(quantity); freePoly(revenue);
    return 0;
}

level: hard
domain: ds
@ statement
كل زبون قد يُحيل زبائن آخرين (رابط إحالة في الاتجاهين = صداقة). نمثل الشبكة بـ **قوائم جوار**: لكل زبون قائمة متصلة بجيرانه. اكتب `addEdge` و **البحث بالعرض** (BFS) بطابور لحساب **درجة الانفصال** بين زبون مصدر وكل الآخرين، وعدد المكونات المتصلة.
@ steps

- BFS: نبدأ من المصدر بمسافة 0، ونعالج الطابور: كل جار غير مزار يأخذ مسافة +1.
- المكونات: نكرر BFS من كل عقدة غير مزارة.
@ flow
for: idx = 0 ; idx < NBE ; idx++
  call: addEdge(edges[idx][0], edges[idx][1])
do: dist[*] = -1 ; dist[source] = 0 ; enqueue(source)
while: الطابور غير فارغ
  do: current = dequeue
  for: edge = adjacency[current] ; edge != NULL ; edge = edge->next
    if: dist[edge->to] == -1
      do: dist[edge->to] = dist[current] + 1 ; enqueue(edge->to)

#include <stdio.h>
#include <stdlib.h>

#define NBN 9

typedef struct Edge { int to; struct Edge *next; } Edge;

Edge *adjacency[NBN];

void addEdge(int from, int to) {
    Edge *edge = malloc(sizeof(Edge));
    edge->to = to; edge->next = adjacency[from]; adjacency[from] = edge;
    edge = malloc(sizeof(Edge));
    edge->to = from; edge->next = adjacency[to]; adjacency[to] = edge;   // الاتجاهان
}

void bfs(int source, int dist[]) {
    int queue[NBN], front = 0, rear = 0, current;
    Edge *edge;
    dist[source] = 0;
    queue[rear++] = source;
    while (front < rear) {
        current = queue[front++];
        for (edge = adjacency[current]; edge != NULL; edge = edge->next)
            if (dist[edge->to] == -1) {
                dist[edge->to] = dist[current] + 1;   // مستوى إضافي
                queue[rear++] = edge->to;
            }
    }
}

int main(void) {
    int edges[9][2] = {{0, 1}, {0, 2}, {1, 3}, {2, 3}, {3, 4}, {4, 5}, {6, 7}, {1, 2}, {5, 2}};
    int dist[NBN], node, components = 0;
    Edge *edge, *next;
    for (node = 0; node < 9; node++) addEdge(edges[node][0], edges[node][1]);
    for (node = 0; node < NBN; node++) dist[node] = -1;
    bfs(0, dist);
    printf("Degrees of separation from customer 0:\n");
    for (node = 1; node < NBN; node++)
        if (dist[node] >= 0) printf("  customer %d: %d\n", node, dist[node]);
        else printf("  customer %d: unreachable\n", node);
    for (node = 0; node < NBN; node++) dist[node] = -1;
    for (node = 0; node < NBN; node++)
        if (dist[node] == -1) { bfs(node, dist); components++; }
    printf("Connected groups: %d\n", components);
    for (node = 0; node < NBN; node++)
        for (edge = adjacency[node]; edge != NULL; edge = next) { next = edge->next; free(edge); }
    return 0;
}

#include <stdio.h>
#include <stdlib.h>

#define CAP 8

typedef struct Node { int code; struct Node *next; } Node;

Node *freeList = NULL;
int mallocCalls = 0;

void initPool(Node *pool) {
    int idx;
    for (idx = 0; idx < CAP - 1; idx++) pool[idx].next = &pool[idx + 1];   // سلسلة حرة
    pool[CAP - 1].next = NULL;
    freeList = pool;
}

Node *allocNode(void) {
    Node *node = freeList;
    if (node != NULL) freeList = node->next;
    return node;                                  // NULL إذا امتلأ المجمع
}

void releaseNode(Node *node) {
    node->next = freeList;                        // تعود إلى القائمة الحرة
    freeList = node;
}

int freeCount(void) { int count = 0; Node *walk; for (walk = freeList; walk != NULL; walk = walk->next) count++; return count; }

Node *cloneList(Node *head) {

    int pos;
    for (walk = head; walk != NULL; walk = copy->next) {          // 1) تشبيك

        for (pos = 0; (copy->name[pos] = walk->name[pos]) != '\0'; pos++) {}

        walk->next = copy->next;
        copy->next = copy->next != NULL ? copy->next->next : NULL;

typedef struct Cat { char name[14]; int depth; struct Cat *next, *child; } Cat;

void flatten(Cat *head) {
    Cat *walk, *tail;
    for (walk = head; walk != NULL; walk = walk->next) {
        if (walk->child == NULL) continue;
        for (tail = walk->child; ; tail = tail->next) {
            tail->depth = walk->depth + 1;             // عمق الأبناء
            if (tail->next == NULL) break;
        }
        tail->next = walk->next;                       // الأبناء قبل الإخوة
        walk->next = walk->child;
        walk->child = NULL;
    }
}

title: إعادة ترتيب العرض بالتناوب
level: hard
domain: ec
@ statement
القائمة مرتبة من الأكثر مبيعًا إلى الأقل `L0, L1, ..., Ln`. لتنويع الصفحة نعيد ترتيبها **بالتناوب** `L0, Ln, L1, Ln-1, L2, ...` في مكانها: نجد المنتصف، نقلب النصف الثاني، ثم ندمج النصفين بالتناوب.
@ steps
- المنتصف بالمؤشرين؛ نقطع بعده.
- قلب النصف الثاني في مكانه.
- الدمج: `first->next = second; second->next = nextFirst;` ونتقدم.
@ flow
do: slow = منتصف القائمة
do: second = reverse(slow->next) ; slow->next = NULL
do: first = head
while: second != NULL
  do: nextFirst = first->next ; nextSecond = second->next
  do: first->next = second ; second->next = nextFirst
  do: first = nextFirst ; second = nextSecond
@ code
#include <stdio.h>
#include <stdlib.h>

typedef struct Node { int rank; struct Node *next; } Node;

Node *reverse(Node *head) {
    Node *prev = NULL, *next;
    while (head != NULL) { next = head->next; head->next = prev; prev = head; head = next; }
    return prev;
}

void reorder(Node *head) {
    Node *slow = head, *fast = head, *first, *second, *nextFirst, *nextSecond;
    if (head == NULL) return;
    while (fast->next != NULL && fast->next->next != NULL) { slow = slow->next; fast = fast->next->next; }
    second = reverse(slow->next);               // النصف الثاني مقلوبًا
    slow->next = NULL;
    for (first = head; second != NULL; first = nextFirst, second = nextSecond) {
        nextFirst = first->next;
        nextSecond = second->next;
        first->next = second;                   // تناوب
        second->next = nextFirst;
    }
}

void runCase(int size) {
    Node *head = NULL, *node, *walk, *next;
    int idx;
    for (idx = size - 1; idx >= 0; idx--) { node = malloc(sizeof(Node)); node->rank = idx; node->next = head; head = node; }
    reorder(head);
    printf("%d items:", size);
    for (walk = head; walk != NULL; walk = walk->next) printf(" L%d", walk->rank);
    printf("\n");
    for (walk = head; walk != NULL; walk = next) { next = walk->next; free(walk); }
}

int main(void) {
    runCase(8);
    runCase(7);
    runCase(2);
    return 0;
}

title: ترتيب مهام تجهيز الطلبية طوبولوجيًا
level: hard
domain: ec
@ statement
تجهيز الطلبية يتكون من مهام مترابطة (لا يمكن التغليف قبل الفحص، ولا الشحن قبل الفوترة والتغليف...). المهام عقد، والاعتماديات أسهم في **قوائم جوار**. اكتب **ترتيبًا طوبولوجيًا** بخوارزمية **كان**: نبدأ بالمهام التي لا تعتمد على شيء، ونزيل أسهمها تدريجيًا. اكشف أيضًا الاعتماديات الدائرية.
@ steps
- `inDegree[task]` = عدد المهام التي يجب أن تسبقها.
- طابور المهام ذات `inDegree == 0`؛ عند إخراج مهمة ننقص `inDegree` لكل تابع لها.
- إذا أخرجنا أقل من كل المهام فهناك حلقة.
@ flow
do: حساب inDegree من قوائم الجوار
do: إدخال كل مهمة inDegree == 0 في الطابور
while: الطابور غير فارغ
  do: task = dequeue ; out: task ; done++
  for: arc = adjacency[task] ; arc != NULL ; arc = arc->next
    do: inDegree[arc->to]--
    if: inDegree[arc->to] == 0
      do: enqueue(arc->to)
if: done < NBT
  out: `"cycle detected"`
@ code
#include <stdio.h>
#include <stdlib.h>

#define NBT 7

typedef struct Arc { int to; struct Arc *next; } Arc;

const char *names[NBT] = {"receive", "check", "pick", "invoice", "pack", "label", "ship"};

int topoSort(Arc *adjacency[], int order[]) {
    int inDegree[NBT] = {0}, queue[NBT], front = 0, rear = 0, task, done = 0;
    Arc *arc;
    for (task = 0; task < NBT; task++)
        for (arc = adjacency[task]; arc != NULL; arc = arc->next) inDegree[arc->to]++;
    for (task = 0; task < NBT; task++) if (inDegree[task] == 0) queue[rear++] = task;
    while (front < rear) {
        task = queue[front++];
        order[done++] = task;
        for (arc = adjacency[task]; arc != NULL; arc = arc->next)
            if (--inDegree[arc->to] == 0) queue[rear++] = arc->to;    // كل ما قبلها أنجز
    }
    return done;
}

void addArc(Arc *adjacency[], int from, int to) {
    Arc *arc = malloc(sizeof(Arc));
    arc->to = to; arc->next = adjacency[from]; adjacency[from] = arc;
}

void runCase(int deps[][2], int nbDeps) {
    Arc *adjacency[NBT] = {NULL}, *arc, *next;
    int order[NBT], done, idx;
    for (idx = 0; idx < nbDeps; idx++) addArc(adjacency, deps[idx][0], deps[idx][1]);
    done = topoSort(adjacency, order);
    for (idx = 0; idx < done; idx++) printf("%s%s", names[order[idx]], idx < done - 1 ? " -> " : "\n");
    if (done < NBT) printf("cycle detected: %d task(s) blocked\n", NBT - done);
    for (idx = 0; idx < NBT; idx++)
        for (arc = adjacency[idx]; arc != NULL; arc = next) { next = arc->next; free(arc); }
}

int main(void) {
    int deps[8][2] = {{0, 1}, {1, 2}, {0, 3}, {2, 4}, {3, 5}, {4, 5}, {5, 6}, {3, 6}};
    int badDeps[8][2] = {{0, 1}, {1, 2}, {0, 3}, {2, 4}, {3, 5}, {4, 5}, {5, 6}, {6, 4}};
    runCase(deps, 8);
    runCase(badDeps, 8);
    return 0;
}

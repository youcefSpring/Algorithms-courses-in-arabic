@ steps
- `count++`، `delta = value - mean`، `mean += delta / count`، `m2 += delta * (value - mean)`.
- التباين = `m2 / count`.
- خوارزمية بمرور واحد وذاكرة ثابتة.

#include <stdio.h>

typedef struct { int count; double mean, m2, low, high; } Running;

void update(Running *stats, double value) {
    double delta = value - stats->mean;
    stats->count++;
    stats->mean += delta / stats->count;
    stats->m2 += delta * (value - stats->mean);       // ويلفورد
    if (stats->count == 1 || value < stats->low) stats->low = value;
    if (stats->count == 1 || value > stats->high) stats->high = value;
}

int main(void) {
    double times[8] = {120, 135, 98, 410, 125, 140, 101, 132};
    Running stats = {0, 0, 0, 0, 0};
    int idx;
    for (idx = 0; idx < 8; idx++) {
        update(&stats, times[idx]);
        printf("n=%d value %3.0f ms -> mean %6.2f var %8.2f\n", stats.count, times[idx],
               stats.mean, stats.m2 / stats.count);
    }
    printf("min %.0f, max %.0f\n", stats.low, stats.high);
    return 0;
}

#include <stdio.h>

#define CAP 4

typedef struct { double data[CAP]; double *next; int count; } Ring;

void addPrice(Ring *ring, double price) {
    *ring->next = price;
    ring->next++;
    if (ring->next == ring->data + CAP) ring->next = ring->data;   // دوران
    if (ring->count < CAP) ring->count++;
}

double average(const Ring *ring) {
    double sum = 0;
    int idx;
    for (idx = 0; idx < ring->count; idx++) sum += ring->data[idx];
    return sum / ring->count;
}

int main(void) {
    double prices[7] = {32000, 33500, 31000, 34500, 30000, 29000, 31500};

    int idx;
    ring.next = ring.data;
    ring.count = 0;
    for (idx = 0; idx < 7; idx++) {
        addPrice(&ring, prices[idx]);
        printf("add %.0f -> last %d avg %.2f\n", prices[idx], ring.count, average(&ring));
    }

#include <stdio.h>

const char *findText(const char *haystack, const char *needle) {
    const char *start, *left, *right;
    for (start = haystack; *start != '\0'; start++) {
        left = start;
        right = needle;
        while (*right != '\0' && *left == *right) { left++; right++; }
        if (*right == '\0') return start;         // كل needle طابقت
    }
    return NULL;
}

typedef struct { char name[8]; int stock; } Product;

void swapBytes(void *first, void *second, int size) {
    unsigned char *left = first, *right = second, tmp;
    int pos;
    for (pos = 0; pos < size; pos++) {               // بايتًا ببايت
        tmp = left[pos];
        left[pos] = right[pos];
        right[pos] = tmp;
    }
}

void reverseAny(void *base, int count, int size) {
    unsigned char *bytes = base;
    int low = 0, high = count - 1;
    while (low < high) {
        swapBytes(bytes + low * size, bytes + high * size, size);
        low++;
        high--;
    }
}

@ statement
اكتب هيكل `Stack` (مؤشر على البيانات، السعة، القمة) ودوال `createStack` و `push` (تضاعف السعة عند الامتلاء) و `pop` و `destroyStack`. استعمله للتحقق من توازن الأقواس في قالب صفحة HTML مبسط.
@ steps
- `push`: إذا `top == capacity` نحجز ضعف الحجم، ننسخ، ونحرر القديم.
- لكل قوس فتح `( [ {` نكدسه؛ لكل غلق نسحب ونتحقق من التطابق.
@ flow

for: pos = 0 ; text[pos] != '\0' ; pos++
  if: قوس فتح
    call: push(&stack, text[pos])
  else:
    if: قوس غلق
      call: open = pop(&stack)
      do: التحقق من التطابق

@ code
#include <stdio.h>
#include <stdlib.h>

typedef struct { char *data; int capacity, top; } Stack;

Stack createStack(int capacity) {
    Stack stack;
    stack.data = malloc(capacity);
    stack.capacity = capacity;
    stack.top = 0;
    return stack;
}

void push(Stack *stack, char value) {
    char *bigger;
    int idx;
    if (stack->top == stack->capacity) {            // نضاعف السعة
        bigger = malloc(2 * stack->capacity);
        for (idx = 0; idx < stack->top; idx++) bigger[idx] = stack->data[idx];
        free(stack->data);
        stack->data = bigger;
        stack->capacity *= 2;
    }
    stack->data[stack->top++] = value;
}

char pop(Stack *stack) { return stack->top > 0 ? stack->data[--stack->top] : '\0'; }
void destroyStack(Stack *stack) { free(stack->data); stack->data = NULL; }

int main(void) {
    char text[] = "{div:[{p:(price*(1+tax))},{ul:[(a),(b),{c:[(d)]}]}]}";
    Stack stack = createStack(2);
    int pos, balanced = 1, deepest = 0;
    char open, close;
    for (pos = 0; text[pos] != '\0' && balanced; pos++) {
        close = text[pos];
        if (close == '(' || close == '[' || close == '{') {
            push(&stack, close);
            if (stack.top > deepest) deepest = stack.top;
        } else if (close == ')' || close == ']' || close == '}') {
            open = pop(&stack);
            balanced = (open == '(' && close == ')') || (open == '[' && close == ']') ||
                       (open == '{' && close == '}');
        }
    }
    if (stack.top != 0) balanced = 0;
    printf("%s, max depth %d, final capacity %d\n", balanced ? "balanced" : "unbalanced",
           deepest, stack.capacity);
    destroyStack(&stack);
    return 0;
}

#include <stdio.h>
#include <stdlib.h>

double **newMatrix(int rows, int cols) {
    double **matrix = malloc(rows * sizeof(double *));
    int row, col;
    for (row = 0; row < rows; row++) {
        matrix[row] = malloc(cols * sizeof(double));
        for (col = 0; col < cols; col++) matrix[row][col] = 0;
    }
    return matrix;
}

void freeMatrix(double **matrix, int rows) {
    int row;
    for (row = 0; row < rows; row++) free(matrix[row]);
    free(matrix);
}

int main(void) {
    double qtyData[3][4] = {{10, 5, 0, 8}, {4, 12, 6, 2}, {7, 0, 9, 5}};
    double costData[4][2] = {{1200, 50}, {3200, 120}, {899, 30}, {4100, 150}};
    double **qty = newMatrix(3, 4), **cost = newMatrix(4, 2), **result = newMatrix(3, 2);
    int row, col, mid;
    for (row = 0; row < 3; row++) for (col = 0; col < 4; col++) qty[row][col] = qtyData[row][col];
    for (row = 0; row < 4; row++) for (col = 0; col < 2; col++) cost[row][col] = costData[row][col];
    for (row = 0; row < 3; row++)
        for (col = 0; col < 2; col++)
            for (mid = 0; mid < 4; mid++)
                result[row][col] += qty[row][mid] * cost[mid][col];   // سطر × عمود
    for (row = 0; row < 3; row++)
        printf("Branch %d: purchase %8.0f DA, transport %6.0f DA\n", row + 1, result[row][0], result[row][1]);
    freeMatrix(qty, 3);
    freeMatrix(cost, 4);
    freeMatrix(result, 3);
    return 0;
}

نريد الإجابة بسرعة عن أسئلة مثل "كم بعنا بين اليوم 5 واليوم 20؟". احجز جدول **مجاميع تراكمية** `prefix[day] = مجموع الأيام 0..day-1` بحجم `n + 1`، واملأه بمؤشرين (قراءة وكتابة)، ثم أجب عن كل استعلام بطرح واحد.

1 30
14 14
@ steps
- `prefix[0] = 0`، و `*(out + 1) = *out + *in`.
- مبيعات المجال `[from, to]` (من 1) = `prefix[to] - prefix[from - 1]`.
- البناء خطي، وكل استعلام بعملية واحدة.
@ flow
do: prefix = malloc(n + 1) ; prefix[0] = 0
for: day = 0 ; day < n ; day++

in: nbQueries

  in: from, to
  out: prefix[to] - prefix[from - 1]
@ code
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int days = 30, day, nbQueries, query, from, to;
    long *sales = malloc(days * sizeof(long)), *prefix = malloc((days + 1) * sizeof(long));
    long *in, *out;
    for (day = 0; day < days; day++) sales[day] = 1000 + (day * 37) % 500;   // بيانات تجريبية
    prefix[0] = 0;
    for (in = sales, out = prefix; in < sales + days; in++, out++)
        *(out + 1) = *out + *in;                     // المجموع التراكمي
    scanf("%d", &nbQueries);
    for (query = 0; query < nbQueries; query++) {
        scanf("%d %d", &from, &to);
        printf("Days %2d..%2d: %ld DA\n", from, to, prefix[to] - prefix[from - 1]);
    }
    free(sales);
    free(prefix);
    return 0;
}

title: كشف تسرب الذاكرة بعدّاد
level: medium
domain: ec
@ statement
اكتب دالتين مغلفتين `trackedMalloc(size)` و `trackedFree(ptr)` تعدّان عدد الكتل المحجوزة والمحررة وعدد البايتات الحية. شغّل محاكاة سلة يُنسى فيها تحرير بعض الكتل، ثم اعرض تقرير **التسرب** في النهاية.
@ steps
- عدادات عامة: `allocations`، `releases`، `liveBytes`.
- لمعرفة حجم الكتلة عند التحرير نحجز `size + sizeof(long)` ونخزن الحجم في بدايتها، ونعيد العنوان بعد الحجم.
- التسرب = `allocations - releases`.
@ flow
for: idx = 0 ; idx < 5 ; idx++
  call: carts[idx] = trackedMalloc(...)
for: idx = 0 ; idx < 5 ; idx++
  if: idx != 1 && idx != 3
    call: trackedFree(carts[idx])
out: allocations, releases, liveBytes
@ code
#include <stdio.h>
#include <stdlib.h>

int allocations = 0, releases = 0;
long liveBytes = 0;

void *trackedMalloc(long size) {
    long *block = malloc(size + sizeof(long));
    if (block == NULL) return NULL;
    *block = size;                         // نخزن الحجم قبل البيانات
    allocations++;
    liveBytes += size;
    return block + 1;                      // المستخدم يرى ما بعد الحجم
}

void trackedFree(void *ptr) {
    long *block = (long *)ptr - 1;         // نرجع إلى بداية الكتلة
    liveBytes -= *block;
    releases++;
    free(block);
}

int main(void) {
    double *carts[5];
    int idx, items;
    for (idx = 0; idx < 5; idx++) {
        items = 3 + idx * 2;
        carts[idx] = trackedMalloc(items * sizeof(double));
        carts[idx][0] = 100.0 * items;
    }
    for (idx = 0; idx < 5; idx++)
        if (idx != 1 && idx != 3)          // نسيان تحرير السلتين 1 و 3
            trackedFree(carts[idx]);
    printf("allocations %d, releases %d\n", allocations, releases);
    printf("LEAK: %d blocks, %ld bytes still allocated\n", allocations - releases, liveBytes);
    trackedFree(carts[1]);
    trackedFree(carts[3]);
    printf("after fix: %ld bytes\n", liveBytes);
    return 0;
}

title: إدراج اسم في قائمة مرتبة ديناميكية
level: medium
domain: ec
@ statement
قائمة أسماء العلامات التجارية جدول ديناميكي من المؤشرات `char **brands` مرتب أبجديًا. اكتب دالة `insertSorted(&brands, &count, name)` تحجز جدولًا أكبر بخانة، وتنسخ المؤشرات مع إدراج الاسم الجديد (نسخة محجوزة) في مكانه الصحيح.
@ stdin
4
Samsung
Apple
Xiaomi
Lenovo
@ steps
- الدالة تغير المؤشر `brands` في `main`، لذلك تستقبل `char ***list`.
- نجد موضع الإدراج بالمقارنة الأبجدية، ثم ننسخ المؤشرات قبله وبعده.
- نحرر الجدول القديم (المؤشرات فقط، لا الأسماء).
@ flow
in: nb
for: idx = 0 ; idx < nb ; idx++
  in: name
  call: insertSorted(&brands, &count, name)
out: brands
do: تحرير الأسماء ثم الجدول
@ code
#include <stdio.h>
#include <stdlib.h>

int compareText(const char *first, const char *second) {
    while (*first != '\0' && *first == *second) { first++; second++; }
    return *first - *second;
}

char *copyText(const char *text) {
    int len = 0, pos;
    char *copy;
    while (text[len] != '\0') len++;
    copy = malloc(len + 1);
    for (pos = 0; pos <= len; pos++) copy[pos] = text[pos];
    return copy;
}

void insertSorted(char ***list, int *count, const char *name) {
    char **bigger = malloc((*count + 1) * sizeof(char *));
    int pos = 0, idx;
    while (pos < *count && compareText((*list)[pos], name) < 0) pos++;   // موضع الإدراج
    for (idx = 0; idx < pos; idx++) bigger[idx] = (*list)[idx];
    bigger[pos] = copyText(name);
    for (idx = pos; idx < *count; idx++) bigger[idx + 1] = (*list)[idx];
    free(*list);                          // الجدول القديم (المؤشرات فقط)
    *list = bigger;
    (*count)++;
}

int main(void) {
    char **brands = NULL, name[30];
    int count = 0, nb, idx;
    scanf("%d", &nb);
    for (idx = 0; idx < nb; idx++) {
        scanf("%29s", name);
        insertSorted(&brands, &count, name);
        printf("after %-8s:", name);
        for (int show = 0; show < count; show++) printf(" %s", brands[show]);
        printf("\n");
    }
    for (idx = 0; idx < count; idx++) free(brands[idx]);
    free(brands);
    return 0;
}

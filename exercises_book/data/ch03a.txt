    // يكفي البحث حتى الجذر: إذا كان number = a × b فأحدهما <= الجذر
    for (divisor = 2; prime && divisor * divisor <= number; divisor++) {
        if (number % divisor == 0)
            prime = 0;            // وجدنا قاسمًا
    }
    if (prime)
        printf("%d is prime\n", number);
    else
        printf("%d is not prime\n", number);

}

title: مدرج تكراري للتقييمات
level: medium
domain: ds
@ statement
اكتب برنامجًا يقرأ `nbReviews` تقييمات (1 إلى 5) ويعد تكرار كل قيمة في خمسة عدادات، ثم يرسم **مدرجًا تكراريًا** نصيًا بالرمز `*` لكل نجمة.
@ stdin
15
5 4 5 3 5 4 4 2 5 1 4 5 3 4 5
@ steps
- خمسة عدادات `c1` إلى `c5` نزيدها بـ `switch`.
- لرسم السطر `stars` نستعمل حلقة داخلية تطبع `*` عددًا من المرات يساوي العداد.
- لتجنب تكرار الكود نختار العداد في كل دورة بـ `switch`.
@ flow
in: nbReviews
for: index = 0 ; index < nbReviews ; index++
  in: rating
  do: زيادة العداد c_r
for: stars = 5 ; stars >= 1 ; stars--
  do: bar = عداد التقييم stars
  for: star = 0 ; star < bar ; star++
    out: `"*"`
@ code
#include <stdio.h>

int main(void) {
    int nbReviews, index, rating, stars, star, bar;
    int c1 = 0, c2 = 0, c3 = 0, c4 = 0, c5 = 0;
    scanf("%d", &nbReviews);
    for (index = 0; index < nbReviews; index++) {
        scanf("%d", &rating);
        switch (rating) {              // نزيد عداد القيمة المقروءة
            case 1: c1++; break;
            case 2: c2++; break;
            case 3: c3++; break;
            case 4: c4++; break;
            case 5: c5++; break;
        }
    }
    for (stars = 5; stars >= 1; stars--) {    // من 5 نجوم إلى نجمة واحدة
        switch (stars) {
            case 5: bar = c5; break;
            case 4: bar = c4; break;
            case 3: bar = c3; break;
            case 2: bar = c2; break;
            default: bar = c1;
        }
        printf("%d stars | ", stars);
        for (star = 0; star < bar; star++)   // حلقة داخلية: طول العمود
            printf("*");
        printf(" (%d)\n", bar);
    }
    return 0;
}

@ steps
- `order[]` يحتوي الزبائن 1..4؛ عند المستوى `depth` نبدل `order[depth]` مع كل عنصر بعده، نستدعي `permute(depth + 1)`، ثم نعيد التبديل (**التراجع**).
- الحالة الأساسية `depth == NB`: نحسب الطول `0 → order[0] → ... → order[3] → 0`.
@ flow

out: bestOrder, bestLength
@ code
#include <stdio.h>

#define NB 4

int dist[NB + 1][NB + 1] = {{0, 12, 18, 9, 22},
                            {12, 0, 7, 14, 11},
                            {18, 7, 0, 10, 6},
                            {9, 14, 10, 0, 15},
                            {22, 11, 6, 15, 0}};
int order[NB] = {1, 2, 3, 4}, bestOrder[NB], bestLength = 100000, tours = 0;

void permute(int depth) {
    int idx, tmp, length;
    if (depth == NB) {                                    // ترتيب كامل
        length = dist[0][order[0]] + dist[order[NB - 1]][0];
        for (idx = 0; idx < NB - 1; idx++) length += dist[order[idx]][order[idx + 1]];
        tours++;
        if (length < bestLength) {
            bestLength = length;
            for (idx = 0; idx < NB; idx++) bestOrder[idx] = order[idx];
        }
        return;
    }
    for (idx = depth; idx < NB; idx++) {
        tmp = order[depth]; order[depth] = order[idx]; order[idx] = tmp;
        permute(depth + 1);
        tmp = order[depth]; order[depth] = order[idx]; order[idx] = tmp;   // تراجع
    }
}

int main(void) {
    int idx;
    permute(0);
    printf("%d tours tested\nBest: 0", tours);
    for (idx = 0; idx < NB; idx++) printf(" -> %d", bestOrder[idx]);
    printf(" -> 0 = %d km\n", bestLength);
    return 0;
}

#include <stdio.h>

#define NB 4

int coins[NB] = {25, 10, 7, 1};
int memo[200];

int minCoins(int amount) {
    int idx, candidate, best = 100000;
    if (amount == 0) return 0;
    if (memo[amount] > 0) return memo[amount];
    for (idx = 0; idx < NB; idx++)
        if (coins[idx] <= amount) {
            candidate = 1 + minCoins(amount - coins[idx]);   // نجرب كل قطعة

    for (idx = 0; idx < NB; idx++)
        while (amount >= coins[idx]) { amount -= coins[idx]; count++; }
    return count;
}

        scanf("%d", &amount);
        printf("%3d DA: optimal %d coins, greedy %d coins\n", amount, minCoins(amount), greedy(amount));

@ steps
- عنصر واحد: هو الأصغر والأكبر (0 مقارنة)؛ عنصران: مقارنة واحدة.
- وإلا: `minMax` للنصف الأيسر والأيمن، ثم مقارنتان للدمج.

#include <stdio.h>

int comparisons = 0;

void minMax(double prices[], int low, int high, double result[]) {
    double left[2], right[2];
    int middle;
    if (low == high) {
        result[0] = result[1] = prices[low];
        return;
    }
    if (high == low + 1) {
        comparisons++;
        if (prices[low] < prices[high]) { result[0] = prices[low]; result[1] = prices[high]; }
        else { result[0] = prices[high]; result[1] = prices[low]; }
        return;
    }
    middle = (low + high) / 2;
    minMax(prices, low, middle, left);          // النصف الأيسر
    minMax(prices, middle + 1, high, right);    // النصف الأيمن
    comparisons += 2;
    result[0] = left[0] < right[0] ? left[0] : right[0];
    result[1] = left[1] > right[1] ? left[1] : right[1];
}

int main(void) {
    double prices[16] = {4500, 1200, 899, 27000, 650, 3200, 12500, 99,
                         7800, 2300, 560, 4100, 1500, 18900, 740, 3100};
    double result[2];
    minMax(prices, 0, 15, result);
    printf("Min %.0f, max %.0f\n", result[0], result[1]);
    printf("Comparisons: %d (simple method: %d)\n", comparisons, 2 * (16 - 1));
    return 0;
}

int depthReached = 0;

double rootBisect(double value, double low, double high) {
    double middle = (low + high) / 2;
    depthReached++;
    if (high - low < 1e-9) return middle;            // دقة كافية
    if (middle * middle > value)
        return rootBisect(value, low, middle);
    return rootBisect(value, middle, high);

#include <stdio.h>

void reverseInPlace(char text[], int left, int right) {
    char tmp;
    if (left >= right) return;
    tmp = text[left];                 // تبديل الطرفين
    text[left] = text[right];
    text[right] = tmp;
    reverseInPlace(text, left + 1, right - 1);
}

int main(void) {
    char text[] = "fast delivery for every order";
    int pos, start = 0, length = 0;
    while (text[length] != '\0') length++;
    reverseInPlace(text, 0, length - 1);
    printf("Whole: %s\n", text);
    reverseInPlace(text, 0, length - 1);          // نعيدها كما كانت
    for (pos = 0; pos <= length; pos++)
        if (text[pos] == ' ' || text[pos] == '\0') {
            reverseInPlace(text, start, pos - 1);  // كل كلمة على حدة
            start = pos + 1;
        }
    printf("Words: %s\n", text);
    return 0;
}

نموذج التكلفة كثير حدود `cost(q) = 0.002q³ - 0.5q² + 60q + 1500`. اكتب دالة عودية `horner(coeffs, degree, value)` تحسب `p(q) = a0 + q·(a1 + q·(a2 + ...))` بعدد ضربات يساوي الدرجة فقط، واحسب التكلفة المتوسطة لعدة كميات.
@ steps

- `horner(coeffs, idx) = coeffs[idx] + value * horner(coeffs, idx + 1)`، والحالة الأساسية `idx == degree`.

@ code
#include <stdio.h>

double horner(double coeffs[], int index, int degree, double value) {
    if (index == degree) return coeffs[degree];
    return coeffs[index] + value * horner(coeffs, index + 1, degree, value);
}

int main(void) {
    double coeffs[4] = {1500, 60, -0.5, 0.002};   // a0 + a1 q + a2 q^2 + a3 q^3
    double quantities[5] = {10, 50, 100, 150, 200}, cost;
    int idx;
    for (idx = 0; idx < 5; idx++) {
        cost = horner(coeffs, 0, 3, quantities[idx]);
        printf("q = %3.0f: cost %9.2f, per unit %7.2f\n",
               quantities[idx], cost, cost / quantities[idx]);
    }
    return 0;
}

#include <stdio.h>

void mergeRec(int first[], int sizeA, int second[], int sizeB, int out[]) {
    int idx;
    if (sizeA == 0) { for (idx = 0; idx < sizeB; idx++) out[idx] = second[idx]; return; }
    if (sizeB == 0) { for (idx = 0; idx < sizeA; idx++) out[idx] = first[idx]; return; }
    if (first[0] <= second[0]) {
        out[0] = first[0];
        mergeRec(first + 1, sizeA - 1, second, sizeB, out + 1);    // باقي الأولى
    } else {
        out[0] = second[0];
        mergeRec(first, sizeA, second + 1, sizeB - 1, out + 1);    // باقي الثانية
    }
}

    int supplierB[6] = {300, 899, 1200, 2500, 4100, 9900};
    int merged[11], idx;
    mergeRec(supplierA, 5, supplierB, 6, merged);
    for (idx = 0; idx < 11; idx++) printf("%d ", merged[idx]);
    printf("\n");
    return 0;
}

level: medium
domain: ds
@ statement
ملف إعدادات المتجر يستعمل الأقواس `{}` و `[]` بشكل متداخل. اكتب دالة عودية `parse(text, pos, depth, closing)` تمر على النص، وعند كل قوس فتح تستدعي نفسها بعمق أكبر، وتعيد الموضع بعد قوس الغلق المطابق. احسب **أقصى عمق** وتحقق من صحة التوازن.
@ steps
- الدالة تعالج كتلة كاملة: تتقدم حتى تجد قوس الغلق المطابق لقوسها.
- عند قوس فتح: `pos = parse(text, pos + 1, depth + 1, closing)` حيث `closing` قوس الغلق المنتظر.
- قوس غلق غير مطابق أو نهاية مبكرة: خطأ (`-1`).
@ flow
call: end = parse(config, 0, 0, '\0')
if: end >= 0
  out: maxDepth
else:
  out: `"unbalanced"`
@ code
#include <stdio.h>

int maxDepth = 0;

int parse(char text[], int pos, int depth, char closing) {
    char ch;
    if (depth > maxDepth) maxDepth = depth;
    while (text[pos] != '\0') {
        ch = text[pos];
        if (ch == '{' || ch == '[') {
            pos = parse(text, pos + 1, depth + 1, ch == '{' ? '}' : ']');   // كتلة داخلية
            if (pos < 0) return -1;
        } else if (ch == '}' || ch == ']') {
            return ch == closing ? pos + 1 : -1;     // يجب أن يطابق قوس الفتح
        } else {
            pos++;
        }
    }
    return closing == '\0' ? pos : -1;               // نهاية النص
}

int main(void) {
    char good[] = "{shop:{name:x,cats:[{id:1,tags:[a,b]},{id:2}]},pay:[cash]}";
    char bad[] = "{shop:{cats:[{id:1]}}";
    int end;
    end = parse(good, 0, 0, '\0');
    printf("Config 1: %s, max depth %d\n", end >= 0 ? "balanced" : "unbalanced", maxDepth);
    maxDepth = 0;
    end = parse(bad, 0, 0, '\0');
    printf("Config 2: %s\n", end >= 0 ? "balanced" : "unbalanced");
    return 0;
}

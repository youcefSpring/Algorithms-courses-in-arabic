#include <stdlib.h>

Tree *newTree(const char *name) {

    int pos;
    for (pos = 0; (node->name[pos] = name[pos]) != '\0'; pos++) {}
    node->left = node->right = NULL;

void printPre(const Tree *node, int depth) {

    freeTree(node->right);

@ statement
**ارتفاع** الشجرة هو عدد العقد في أطول مسار من الجذر إلى ورقة (عدد مستويات القائمة المنسدلة). اكتب `height(node)` العودية = 1 + الأكبر بين ارتفاعي الفرعين.

@ flow

else:

  call: rightH = height(node->right)

#include <stdio.h>
#include <stdlib.h>

typedef struct Tree { int code; struct Tree *left, *right; } Tree;

    node->code = code; node->left = left; node->right = right;

int height(const Tree *node) {
    int leftH, rightH;
    if (node == NULL) return 0;
    leftH = height(node->left);
    rightH = height(node->right);
    return 1 + (leftH > rightH ? leftH : rightH);
}

void freeTree(Tree *node) { if (node) { freeTree(node->left); freeTree(node->right); free(node); } }

}

title: البحث عن رمز في شجرة البحث
level: easy
domain: ec
@ statement
اكتب `search(node, code, &steps)` **تكرارية** (دون عودية) تبحث عن رمز منتج في شجرة بحث: في كل عقدة نذهب يسارًا أو يمينًا فقط. اعرض عدد العقد المزارة لكل بحث.
@ stdin

@ steps
- ما دام `node != NULL`: إذا ساوى الرمز نعيد العقدة، وإلا نختار الفرع.
- عدد الخطوات لا يتجاوز ارتفاع الشجرة.
@ flow
in: code
do: node = root ; steps = 0
while: node != NULL && node->code != code
  do: steps++
  if: code < node->code
    do: node = node->left
  else:
    do: node = node->right
out: node != NULL, steps
@ code
#include <stdio.h>
#include <stdlib.h>

typedef struct Tree { int code; struct Tree *left, *right; } Tree;

Tree *insert(Tree *node, int code) {
    if (node == NULL) { node = malloc(sizeof(Tree)); node->code = code; node->left = node->right = NULL; }
    else if (code < node->code) node->left = insert(node->left, code);
    else node->right = insert(node->right, code);
    return node;
}

Tree *search(Tree *node, int code, int *steps) {
    *steps = 0;
    while (node != NULL) {
        (*steps)++;
        if (code == node->code) return node;
        node = code < node->code ? node->left : node->right;   // فرع واحد فقط
    }
    return NULL;
}

void freeTree(Tree *node) { if (node) { freeTree(node->left); freeTree(node->right); free(node); } }

int main(void) {
    int codes[9] = {5120, 2045, 7781, 1001, 3310, 6400, 9002, 1500, 4400}, idx, code, steps;
    Tree *root = NULL;
    for (idx = 0; idx < 9; idx++) root = insert(root, codes[idx]);
    for (idx = 0; idx < 2; idx++) {
        scanf("%d", &code);
        printf("Code %d: %s after %d node(s)\n", code,
               search(root, code, &steps) != NULL ? "found" : "not found", steps);
    }
    freeTree(root);
    return 0;
}

#include <stdlib.h>

Tree *insert(Tree *node, int price) {
    if (node == NULL) { node = malloc(sizeof(Tree)); node->price = price; node->left = node->right = NULL; }
    else if (price < node->price) node->left = insert(node->left, price);
    else node->right = insert(node->right, price);
    return node;
}

Tree *removeKey(Tree *node, int price) {
    Tree *child, *succ;
    if (node == NULL) return NULL;
    if (price < node->price) node->left = removeKey(node->left, price);
    else if (price > node->price) node->right = removeKey(node->right, price);
    else if (node->left == NULL || node->right == NULL) {      // 0 أو 1 ابن
        child = node->left != NULL ? node->left : node->right;
        free(node);
        return child;
    } else {                                                   // ابنان
        for (succ = node->right; succ->left != NULL; succ = succ->left) {}
        node->price = succ->price;                             // نسخ الخلف
        node->right = removeKey(node->right, succ->price);
    }
    return node;
}

void freeTree(Tree *node) { if (node) { freeTree(node->left); freeTree(node->right); free(node); } }

        if (first < node->price && second < node->price) node = node->left;
        else if (first > node->price && second > node->price) node = node->right;
        else return node;                             // تنفصل القيمتان هنا

const Tree *steps[10];
char answers[10];
int ruleCount = 0;

void printRules(const Tree *node, int depth) {
    int idx;
    if (node->left == NULL && node->right == NULL) {          // ورقة: قرار
        printf("R%d: IF", ++ruleCount);
        for (idx = 0; idx < depth; idx++)
            printf("%s %s%s", idx ? " AND" : "", answers[idx] == 'N' ? "NOT " : "", steps[idx]->text);

    answers[depth] = 'Y';
    printRules(node->left, depth + 1);
    answers[depth] = 'N';
    printRules(node->right, depth + 1);
}

@ statement
لدينا كتالوج **مرتب** حسب الرمز. الإدراج المتتالي يعطي سلسلة. اكتب `buildBalanced(codes, low, high)` التي تجعل العنصر الأوسط جذرًا، وتبني الفرعين عوديًا من النصفين، فتعطي شجرة **بأقل ارتفاع ممكن**. قارن الارتفاع بالبناء بالإدراج.
@ steps
- `middle = (low + high) / 2` يصبح الجذر.
- اليسار من `[low, middle-1]` واليمين من `[middle+1, high]`.
- الارتفاع ≈ `log2(n) + 1`.
@ flow
if: low > high
  out: NULL
else:
  do: middle = (low + high) / 2 ; node = newTree(codes[middle])
  call: node->left = buildBalanced(codes, low, middle - 1)
  call: node->right = buildBalanced(codes, middle + 1, high)
  out: node
@ code
#include <stdio.h>
#include <stdlib.h>

typedef struct Tree { int code; struct Tree *left, *right; } Tree;

Tree *newTree(int code) {
    Tree *node = malloc(sizeof(Tree));
    node->code = code; node->left = node->right = NULL;
    return node;
}

Tree *buildBalanced(const int *codes, int low, int high) {
    Tree *node;
    int middle;
    if (low > high) return NULL;
    middle = (low + high) / 2;
    node = newTree(codes[middle]);                       // الأوسط جذرًا
    node->left = buildBalanced(codes, low, middle - 1);
    node->right = buildBalanced(codes, middle + 1, high);
    return node;
}

Tree *insert(Tree *node, int code) {
    if (node == NULL) return newTree(code);
    if (code < node->code) node->left = insert(node->left, code);
    else node->right = insert(node->right, code);
    return node;
}

int height(const Tree *node) {
    int leftH, rightH;
    if (node == NULL) return 0;
    leftH = height(node->left); rightH = height(node->right);
    return 1 + (leftH > rightH ? leftH : rightH);
}

void freeTree(Tree *node) { if (node) { freeTree(node->left); freeTree(node->right); free(node); } }

int main(void) {
    int codes[15], idx;
    Tree *balanced, *chain = NULL;
    for (idx = 0; idx < 15; idx++) codes[idx] = 1001 + idx * 7;
    balanced = buildBalanced(codes, 0, 14);
    for (idx = 0; idx < 15; idx++) chain = insert(chain, codes[idx]);
    printf("15 sorted codes\nbalanced build: root %d, height %d\n", balanced->code, height(balanced));
    printf("insert one by one: root %d, height %d\n", chain->code, height(chain));
    freeTree(balanced);
    freeTree(chain);
    return 0;
}

صيغة السعر `(price - discount) * qty + shipping` تُمثل **بشجرة عبارة**: العقد الداخلية عمليات (`+ - * /`) والأوراق أعداد. اكتب `evaluate(node)` العودية، و `printInfix(node)` التي تعيد كتابة العبارة بالأقواس، ثم احسب السعر لعدة كميات بتغيير ورقة واحدة.
@ steps
- ورقة: نعيد قيمتها.
- عقدة عملية: نحسب الفرعين ثم نطبق العملية.
- `printInfix`: `(` يسار عملية يمين `)`.
@ flow
if: node->op == 0
  out: node->value
else:
  call: left = evaluate(node->left)
  call: right = evaluate(node->right)
  out: left op right
@ code
#include <stdio.h>
#include <stdlib.h>

typedef struct Tree { char op; double value; struct Tree *left, *right; } Tree;

Tree *leaf(double value) {
    Tree *node = malloc(sizeof(Tree));
    node->op = 0; node->value = value; node->left = node->right = NULL;
    return node;
}

Tree *operation(char op, Tree *left, Tree *right) {
    Tree *node = malloc(sizeof(Tree));
    node->op = op; node->value = 0; node->left = left; node->right = right;
    return node;
}

double evaluate(const Tree *node) {
    double left, right;
    if (node->op == 0) return node->value;                 // ورقة
    left = evaluate(node->left);
    right = evaluate(node->right);
    switch (node->op) {
        case '+': return left + right;
        case '-': return left - right;
        case '*': return left * right;
        default:  return left / right;
    }
}

void printInfix(const Tree *node) {
    if (node->op == 0) { printf("%g", node->value); return; }
    printf("(");
    printInfix(node->left);
    printf(" %c ", node->op);
    printInfix(node->right);
    printf(")");
}

void freeTree(Tree *node) { if (node) { freeTree(node->left); freeTree(node->right); free(node); } }

int main(void) {
    Tree *qty = leaf(1);
    Tree *root = operation('+', operation('*', operation('-', leaf(4500), leaf(500)), qty), leaf(400));
    int count;
    printInfix(root);
    printf("\n");
    for (count = 1; count <= 4; count++) {
        qty->value = count;                                    // نغير الورقة فقط
        printf("qty %d -> %.0f DA\n", count, evaluate(root));
    }
    freeTree(root);
    return 0;
}

@ steps

- البادئة: نزور عقدة إذا كان مجال اسمها يمكن أن يحتوي على البادئة؛ المرور الوسطي يعطي النتائج مرتبة.

@ code
#include <stdio.h>
#include <stdlib.h>

typedef struct Tree { char name[16]; double price; struct Tree *left, *right; } Tree;

int compareText(const char *first, const char *second) {
    while (*first != '\0' && *first == *second) { first++; second++; }
    return *first - *second;
}

int startsWith(const char *text, const char *prefix) {
    while (*prefix != '\0') if (*text++ != *prefix++) return 0;
    return 1;
}

Tree *insert(Tree *node, const char *name, double price) {
    int pos;
    if (node == NULL) {
        node = malloc(sizeof(Tree));
        for (pos = 0; (node->name[pos] = name[pos]) != '\0'; pos++) {}
        node->price = price;
        node->left = node->right = NULL;
    } else if (compareText(name, node->name) < 0) node->left = insert(node->left, name, price);
    else node->right = insert(node->right, name, price);
    return node;
}

void inorder(const Tree *node) { if (node) { inorder(node->left); printf("%s ", node->name); inorder(node->right); } }

void printPrefix(const Tree *node, const char *prefix) {
    if (node == NULL) return;
    if (compareText(prefix, node->name) < 0) printPrefix(node->left, prefix);    // قد توجد يسارًا
    if (startsWith(node->name, prefix)) printf("  %s %.0f\n", node->name, node->price);
    if (startsWith(node->name, prefix) || compareText(node->name, prefix) < 0) printPrefix(node->right, prefix);
}

void freeTree(Tree *node) { if (node) { freeTree(node->left); freeTree(node->right); free(node); } }

int main(void) {
    const char *names[8] = {"Monitor", "Cable", "Mouse", "Laptop", "Webcam", "Modem", "Hub", "Keyboard"};
    double prices[8] = {32000, 450, 1500, 89000, 4100, 6200, 3100, 3200};
    Tree *root = NULL;
    char prefix[16];
    int idx;
    for (idx = 0; idx < 8; idx++) root = insert(root, names[idx], prices[idx]);
    printf("A-Z: "); inorder(root); printf("\n");
    scanf("%15s", prefix);
    printf("Starting with \"%s\":\n", prefix);
    printPrefix(root, prefix);
    freeTree(root);
    return 0;
}

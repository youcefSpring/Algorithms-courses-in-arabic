#include <stdio.h>

#define NB 10

int buffer[NB];

long sortCount(int values[], int low, int high) {
    int middle, first, second, out, idx;
    long count;
    if (low >= high) return 0;
    middle = (low + high) / 2;
    count = sortCount(values, low, middle) + sortCount(values, middle + 1, high);
    first = low; second = middle + 1; out = low;
    while (first <= middle && second <= high) {
        if (values[first] <= values[second]) {
            buffer[out++] = values[first++];
        } else {
            buffer[out++] = values[second++];
            count += middle - first + 1;          // كل الباقي في اليسار أكبر منه
        }

    while (first <= middle) buffer[out++] = values[first++];
    while (second <= high) buffer[out++] = values[second++];
    for (idx = low; idx <= high; idx++) values[idx] = buffer[idx];
    return count;
}

void quickSort(int prices[], int low, int high) {
    int pivot, left = low, right = high, tmp;

    quickSort(prices, left, high);             // الجزء الأكبر

        diff = cols[prev] > col ? cols[prev] - col : col - cols[prev];

        }
}

وسّع حاسبة صيغ الأسعار لتقبل **الأقواس** والأعداد العشرية: `(1200+800)*1.19-(2*150)/3`. اكتب **محللًا تنازليًا عوديًا** من ثلاث دوال متبادلة: `expression` (جمع وطرح)، `term` (ضرب وقسمة)، `factor` (عدد أو عبارة بين قوسين).

@ steps
- `expression := term (('+'|'-') term)*`.
- `term := factor (('*'|'/') factor)*`.
- `factor := number | '(' expression ')'`: هنا تحدث العودية غير المباشرة.
- موضع القراءة متغير عام `pos` تتقدم فيه كل الدوال.

@ code
#include <stdio.h>

char formula[100];
int pos = 0;

double expression(void);

double factor(void) {
    double value = 0, scale = 1;
    if (formula[pos] == '(') {
        pos++;                                     // (
        value = expression();                      // عودية غير مباشرة
        pos++;                                     // )
        return value;
    }
    while (formula[pos] >= '0' && formula[pos] <= '9')
        value = value * 10 + (formula[pos++] - '0');
    if (formula[pos] == '.') {
        pos++;
        while (formula[pos] >= '0' && formula[pos] <= '9') {
            scale /= 10;
            value += (formula[pos++] - '0') * scale;
        }
    }
    return value;
}

double term(void) {
    double value = factor();
    while (formula[pos] == '*' || formula[pos] == '/') {
        if (formula[pos++] == '*') value *= factor();
        else value /= factor();
    }
    return value;
}

double expression(void) {
    double value = term();
    while (formula[pos] == '+' || formula[pos] == '-') {
        if (formula[pos++] == '+') value += term();
        else value -= term();
    }
    return value;
}

int main(void) {
    scanf("%99s", formula);
    printf("%s = %.2f DA\n", formula, expression());
    return 0;
}

#define NB 15

int calls = 0;

int partition(double values[], int low, int high) {
    double pivot = values[high], tmp;
    int store = low, idx;
    for (idx = low; idx < high; idx++)
        if (values[idx] < pivot) {
            tmp = values[idx]; values[idx] = values[store]; values[store] = tmp;
            store++;

    tmp = values[store]; values[store] = values[high]; values[high] = tmp;
    return store;                                  // الموضع النهائي للمحور

    if (rank < pivotPos) return quickSelect(values, low, pivotPos - 1, rank);
    return quickSelect(values, pivotPos + 1, high, rank);   // جهة واحدة فقط
}

#include <stdio.h>

#define NB 12

double xs[NB] = {2, 15, 7, 31, 22, 9, 40, 18, 27, 4, 35, 12};
double ys[NB] = {3, 8, 14, 5, 19, 2, 11, 25, 9, 20, 16, 13};
int pairA, pairB;

double dist2(int first, int second) {
    return (xs[first] - xs[second]) * (xs[first] - xs[second])
         + (ys[first] - ys[second]) * (ys[first] - ys[second]);
}

double check(int first, int second, double best) {
    double value = dist2(first, second);
    if (value < best) { pairA = first; pairB = second; return value; }
    return best;
}

double closest(int low, int high) {
    double best = 1e18, right, gap;
    int first, second, middle;
    if (high - low < 3) {                                   // حالة أساسية صغيرة
        for (first = low; first <= high; first++)
            for (second = first + 1; second <= high; second++)
                best = check(first, second, best);
        return best;
    }
    middle = (low + high) / 2;
    best = closest(low, middle);
    right = closest(middle + 1, high);
    if (right < best) best = right;
    for (first = low; first <= high; first++) {             // الشريط حول خط القسمة
        gap = xs[first] - xs[middle];
        if (gap * gap >= best) continue;
        for (second = first + 1; second <= high; second++) {
            gap = xs[second] - xs[middle];
            if (gap * gap < best) best = check(first, second, best);
        }
    }
    return best;
}

double mySqrt(double value) {
    double guess = value > 1 ? value : 1;
    int iter;
    for (iter = 0; iter < 50; iter++) guess = (guess + value / guess) / 2;
    return guess;
}

           xs[pairA], ys[pairA], xs[pairB], ys[pairB]);
    printf("Distance: %.3f km\n", mySqrt(best));

                    {100, 72, 81, 92, 94, 95, 0}};
int visited[NB], path[NB], bestPath[NB], best = 1000000;
long nodes = 0;

void explore(int city, int visitedCount, int length) {
    int next, idx;
    nodes++;
    if (length >= best) return;                        // تحديد: لا فائدة من المواصلة
    if (visitedCount == NB) {
        if (length + dist[city][0] < best) {
            best = length + dist[city][0];
            for (idx = 0; idx < NB; idx++) bestPath[idx] = path[idx];
        }
        return;
    }
    for (next = 1; next < NB; next++)
        if (!visited[next]) {
            visited[next] = 1;
            path[visitedCount] = next;
            explore(next, visitedCount + 1, length + dist[city][next]);
            visited[next] = 0;                         // تراجع
        }
}

int main(void) {
    int idx;
    path[0] = 0;
    visited[0] = 1;
    explore(0, 1, 0);
    printf("Best tour:");
    for (idx = 0; idx < NB; idx++) printf(" %d ->", bestPath[idx]);
    printf(" 0\nLength: %d km\n", best);
    printf("Nodes explored: %ld (full tree: 1957)\n", nodes);
    return 0;
}

- الحالة الأساسية: `start == length` → نجاح.

@ code
#include <stdio.h>

#define NBW 12

char dict[NBW][10] = {"free", "shipping", "ship", "ping", "for", "summer", "sale",
                      "sum", "mer", "fors", "um", "s"};
char tag[60];
int length, choice[60];
long ways[60];

int matchAt(int start, int word) {
    int pos = 0;
    while (dict[word][pos] != '\0') {
        if (tag[start + pos] != dict[word][pos]) return 0;
        pos++;
    }
    return pos;                                      // طول الكلمة المطابقة
}

long countWays(int start) {
    int word, len;
    long total = 0, sub;
    if (start == length) return 1;
    if (ways[start] >= 0) return ways[start];
    choice[start] = -1;
    for (word = 0; word < NBW; word++) {
        len = matchAt(start, word);
        if (len > 0) {
            sub = countWays(start + len);
            if (sub > 0 && choice[start] < 0) choice[start] = word;
            total += sub;
        }
    }
    ways[start] = total;
    return total;
}

int main(void) {
    int start;
    long total;
    scanf("%59s", tag);
    for (length = 0; tag[length] != '\0'; length++) ways[length] = -1;
    total = countWays(0);
    printf("#%s: %ld possible splits\n", tag, total);
    if (total > 0) {
        printf("First split:");
        for (start = 0; start < length; start += matchAt(start, choice[start]))
            printf(" %s", dict[choice[start]]);
        printf("\n");
    }
    return 0;
}

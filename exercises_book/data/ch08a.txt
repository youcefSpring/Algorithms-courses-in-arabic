@ statement
حساب `base^exponent` بالضرب المتكرر يتطلب `exponent` عملية. اكتب دالة عودية `fastPower` تستعمل: `base^n = (base^(n/2))²` إذا كان `n` زوجيًا، و `base × base^(n-1)` إذا كان فرديًا، وعدّ عدد عمليات الضرب.
@ steps
- نحسب `half = fastPower(base, n / 2)` **مرة واحدة** ثم نعيد `half * half`.
- عدد الضربات من رتبة `log2(n)`.
- نستعمل جدولًا من خانة واحدة لعد الضربات.
@ flow
call: result = fastPower(1.01, exponent, counter)
out: result, counter[0]
@ code
#include <stdio.h>

double fastPower(double base, int exponent, int counter[]) {
    double half;
    if (exponent == 0) return 1;
    if (exponent % 2 == 1) {
        counter[0]++;
        return base * fastPower(base, exponent - 1, counter);   // أس فردي
    }
    half = fastPower(base, exponent / 2, counter);              // نحسب النصف مرة واحدة
    counter[0]++;
    return half * half;
}

int main(void) {
    int exponents[4] = {10, 100, 365, 1000}, counter[1], idx;
    double result;
    for (idx = 0; idx < 4; idx++) {
        counter[0] = 0;
        result = fastPower(1.01, exponents[idx], counter);
        printf("1.01^%-4d = %14.4f with %2d multiplications (naive: %d)\n",
               exponents[idx], result, counter[0], exponents[idx]);
    }
    return 0;
}

int isPalindrome(char text[], int left, int right) {
    if (left >= right) return 1;                     // انتهت المقارنات

               isPalindrome(codes[idx], 0, length - 1) ? "palindrome" : "no");

#include <stdio.h>

int moves = 0;

void hanoi(int count, char from, char to, char via) {
    if (count == 0) return;
    hanoi(count - 1, from, via, to);           // نفرغ الطريق
    moves++;
    printf("Move %d: box %d from %c to %c\n", moves, count, from, to);
    hanoi(count - 1, via, to, from);           // نعيد الكومة فوقه
}

int main(void) {
    int count;
    scanf("%d", &count);
    hanoi(count, 'A', 'C', 'B');
    printf("Total moves: %d\n", moves);
    return 0;

title: فيبوناتشي بالتذكر وعدد الاستدعاءات
level: medium
domain: ds
@ statement
الدالة العودية الساذجة `fib` تعيد حساب نفس القيم آلاف المرات. اكتب نسخة **بالتذكر (memoization)** تحفظ كل نتيجة في جدول، وقارن عدد الاستدعاءات بين النسختين لـ `fib(30)`.
@ steps
- جدول `memo[]` مهيأ بـ 0 (= لم يُحسب بعد).
- قبل الحساب: إذا `memo[month] != 0` نعيدها مباشرة.
- عداد عام لكل نسخة.
@ flow
call: slowValue = fibSlow(30)
call: fastValue = fibMemo(30)
out: slowValue, slowCalls
out: fastValue, fastCalls
@ code
#include <stdio.h>

long slowCalls = 0, fastCalls = 0;
long memo[100];

long fibSlow(int month) {
    slowCalls++;
    if (month <= 2) return 1;
    return fibSlow(month - 1) + fibSlow(month - 2);
}

long fibMemo(int month) {
    fastCalls++;
    if (month <= 2) return 1;
    if (memo[month] != 0) return memo[month];          // محسوبة سابقًا
    memo[month] = fibMemo(month - 1) + fibMemo(month - 2);
    return memo[month];
}

int main(void) {
    long slowValue = fibSlow(30), fastValue = fibMemo(30);
    printf("Naive:    fib(30) = %ld with %ld calls\n", slowValue, slowCalls);
    printf("Memoized: fib(30) = %ld with %ld calls\n", fastValue, fastCalls);
    fastCalls = 0;
    printf("fib(90) = %ld\n", fibMemo(90));
    printf("calls for fib(90) after cache: %ld\n", fastCalls);
    return 0;
}

title: عدد الحزم الممكنة بمثلث باسكال
level: medium
domain: ec
@ statement
بكم طريقة نختار `pick` منتجات من `total` منتج لتكوين حزمة؟ اكتب الدالة العودية `choose(total, pick) = choose(total-1, pick-1) + choose(total-1, pick)` مع التذكر في جدول ثنائي البعد، واعرض الأسطر الأولى من **مثلث باسكال**.
@ steps
- الحالتان الأساسيتان: `pick == 0` أو `pick == total` → 1.
- إما أن المنتج الأخير في الحزمة (`pick - 1` من الباقي)، أو لا (`pick` من الباقي).
- `memo[total][pick]` يمنع تكرار الحساب.
@ flow
for: total = 0 ; total <= 6 ; total++
  for: pick = 0 ; pick <= total ; pick++
    out: choose(total, pick)
call: bundles = choose(20, 3)
out: bundles
@ code
#include <stdio.h>

long memo[60][60];

long choose(int total, int pick) {
    if (pick == 0 || pick == total) return 1;
    if (memo[total][pick] != 0) return memo[total][pick];
    memo[total][pick] = choose(total - 1, pick - 1) + choose(total - 1, pick);
    return memo[total][pick];
}

int main(void) {
    int total, pick;
    for (total = 0; total <= 6; total++) {
        printf("%*s", 2 * (6 - total), "");         // إزاحة لشكل المثلث
        for (pick = 0; pick <= total; pick++)
            printf("%4ld", choose(total, pick));
        printf("\n");
    }
    printf("Bundles of 3 among 20 products: %ld\n", choose(20, 3));
    printf("Bundles of 10 among 50 products: %ld\n", choose(50, 10));
    return 0;
}

title: كل الحزم الممكنة من أربعة منتجات
level: medium
domain: ec
@ statement
اكتب دالة عودية `subsets(index, chosen)` تولّد **كل المجموعات الجزئية** لأربعة منتجات (16 حزمة)، وتعرض لكل حزمة محتواها وسعرها الإجمالي، وعدد الحزم التي لا يتجاوز سعرها 6000 دج.
@ steps
- لكل منتج قراران: نأخذه (`chosen[index] = 1`) أو لا نأخذه، ثم ننتقل إلى المنتج التالي.
- الحالة الأساسية: `index == NB` → حزمة كاملة نعرضها.
- عدد الحزم `2^NB`.
@ flow
call: subsets(0)
@ code
#include <stdio.h>

#define NB 4

char names[NB][10] = {"Mouse", "Keyboard", "Headset", "Webcam"};
double prices[NB] = {1500, 3200, 2800, 4100};
int chosen[NB], affordable = 0;

void subsets(int index) {
    int idx;
    double total = 0;
    if (index == NB) {                         // حزمة كاملة
        printf("{");
        for (idx = 0; idx < NB; idx++)
            if (chosen[idx]) { printf(" %s", names[idx]); total += prices[idx]; }
        printf(" } = %.0f\n", total);
        if (total <= 6000) affordable++;
        return;
    }
    chosen[index] = 0;                         // دون المنتج
    subsets(index + 1);
    chosen[index] = 1;                         // مع المنتج
    subsets(index + 1);
}

int main(void) {
    subsets(0);
    printf("Bundles up to 6000 DA: %d\n", affordable);
    return 0;
}

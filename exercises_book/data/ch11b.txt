- `replay` تقرأ كل الأسطر وتطبقها بالترتيب.

#include <stdio.h>

void appendOp(char type, double amount) {
    FILE *file = fopen("ledger.txt", "a");
    fprintf(file, "%c %.2f\n", type, amount);
    fclose(file);
}

double replay(int *count) {
    FILE *file = fopen("ledger.txt", "r");
    char type;
    double amount, balance = 0;
    *count = 0;
    if (file == NULL) return 0;
    while (fscanf(file, " %c %lf", &type, &amount) == 2) {
        if (type == 'D') balance += amount;
        else balance -= amount;                      // W أو F
        (*count)++;
    }
    fclose(file);
    return balance;
}

    FILE *file = fopen("ledger.txt", "w");          // دفتر جديد
    int count;
    double balance;
    fclose(file);
    appendOp('D', 5000); appendOp('W', 1200); appendOp('F', 50);
    balance = replay(&count);
    printf("After %d ops: %.2f DA\n", count, balance);
    appendOp('D', 2500); appendOp('W', 4000);
    balance = replay(&count);
    printf("After %d ops: %.2f DA\n", count, balance);
    return 0;
}

@ steps
- XOR مرتين بنفس المفتاح يعيد القيمة الأصلية.
- نقرأ ونكتب بايتًا ببايت بالوضعين `"rb"` و `"wb"`.

@ flow

@ code
#include <stdio.h>

void xorFile(const char *source, const char *target, const char *key) {
    FILE *input = fopen(source, "rb"), *output = fopen(target, "wb");
    int ch, keyLen = 0;
    long pos = 0;
    while (key[keyLen] != '\0') keyLen++;
    while ((ch = fgetc(input)) != EOF)
        fputc(ch ^ key[pos++ % keyLen], output);     // المفتاح يتكرر
    fclose(input);
    fclose(output);
}

int compareFiles(const char *first, const char *second) {
    FILE *fileA = fopen(first, "rb"), *fileB = fopen(second, "rb");
    int chA, chB, same = 1;
    do {
        chA = fgetc(fileA);
        chB = fgetc(fileB);
        if (chA != chB) same = 0;
    } while (chA != EOF && chB != EOF && same);
    fclose(fileA);
    fclose(fileB);
    return same;
}

int main(void) {
    FILE *file = fopen("clients.txt", "w");
    int ch, shown = 0;
    fprintf(file, "507;Amina;0550123456\n508;Karim;0661987654\n");
    fclose(file);
    xorFile("clients.txt", "clients.enc", "K3y!");
    printf("Encrypted bytes (hex): ");
    file = fopen("clients.enc", "rb");
    while ((ch = fgetc(file)) != EOF && shown++ < 12) printf("%02X ", ch);
    fclose(file);
    xorFile("clients.enc", "clients.dec", "K3y!");    // نفس العملية تفك التشفير
    printf("...\nDecrypted identical to original: %s\n", compareFiles("clients.txt", "clients.dec") ? "yes" : "no");
    return 0;
}

#include <stdio.h>

char lower(char ch) { return (ch >= 'A' && ch <= 'Z') ? ch - 'A' + 'a' : ch; }

int containsIgnoreCase(const char *line, const char *word) {
    const char *start, *left, *right;
    for (start = line; *start != '\0'; start++) {
        for (left = start, right = word; *right != '\0' && lower(*left) == lower(*right); left++, right++) {}
        if (*right == '\0') return 1;
    }
    return 0;
}

int main(void) {
    FILE *file = fopen("support.log", "w");
    char word[30], line[120];
    int number = 0, matches = 0;
    fprintf(file, "#101 customer asks about delivery time\n#102 REFUND requested: item broken\n");
    fprintf(file, "#103 password reset\n#104 partial Refund approved for #102\n#105 invoice copy\n");
    fprintf(file, "#106 refunds policy question\n");
    fclose(file);
    scanf("%29s", word);
    file = fopen("support.log", "r");
    while (fgets(line, 120, file) != NULL) {
        number++;
        if (containsIgnoreCase(line, word)) {
            printf("%d: %s", number, line);
            matches++;
        }
    }
    fclose(file);
    printf("%d matching line(s) for \"%s\"\n", matches, word);
    return 0;
}

    fclose(file);

int sameText(const char *first, const char *second) {
    while (*first != '\0' && *first == *second) { first++; second++; }
    return *first == *second;
}

void copyText(char *target, const char *source) { while ((*target++ = *source++) != '\0') {} }

int main(void) {
    FILE *file = fopen("timeline.log", "w"), *out;
    char day[12], time[8], currentDay[12] = "", bestDay[12] = "";
    double amount, total = 0, best = -1;
    int hasDay = 0;

level: medium
domain: ec
@ statement
عند إغلاق التطبيق نحفظ السلة في ملف ثنائي: **رأس** (رقم إصدار الصيغة وعدد العناصر) ثم العناصر. عند الفتح نستعيدها بعد التحقق من الإصدار. اكتب `saveCart` و `loadCart` واختبر الحفظ والاستعادة، ثم محاولة تحميل ملف بإصدار قديم.
@ steps
- الرأس هيكل `Header { int version; int count; }`.
- `loadCart` ترفض الملف إذا اختلف الإصدار أو إذا كان العدد غير منطقي.
- نكتب الرأس ثم العناصر بـ `fwrite`.
@ flow

call: count = loadCart("cart.bin", restored)
out: restored
do: كتابة ملف بإصدار 1
call: loadCart("old.bin", restored)
out: رفض
@ code
#include <stdio.h>

#define VERSION 2

typedef struct { int version; int count; } Header;
typedef struct { int code; int qty; double price; } Item;

void saveCart(const char *path, const Item *items, int count) {
    FILE *file = fopen(path, "wb");
    Header header = {VERSION, count};
    fwrite(&header, sizeof(Header), 1, file);          // الرأس أولًا
    fwrite(items, sizeof(Item), count, file);
    fclose(file);
}

int loadCart(const char *path, Item *items, int capacity) {
    FILE *file = fopen(path, "rb");
    Header header;
    int count;
    if (file == NULL) return -1;
    if (fread(&header, sizeof(Header), 1, file) != 1 || header.version != VERSION
        || header.count < 0 || header.count > capacity) {
        fclose(file);
        return -1;                                     // ملف غير متوافق
    }
    count = (int)fread(items, sizeof(Item), header.count, file);
    fclose(file);
    return count;
}

int main(void) {
    Item cart[3] = {{101, 2, 1500}, {205, 1, 32000}, {330, 4, 450}}, restored[10];
    Header oldHeader = {1, 3};
    FILE *file;
    int count, idx;
    saveCart("cart.bin", cart, 3);
    count = loadCart("cart.bin", restored, 10);
    printf("Restored %d items:", count);
    for (idx = 0; idx < count; idx++) printf(" %d x%d", restored[idx].code, restored[idx].qty);
    printf("\n");
    file = fopen("old.bin", "wb");
    fwrite(&oldHeader, sizeof(Header), 1, file);
    fwrite(cart, sizeof(Item), 3, file);
    fclose(file);
    printf("Old format file: %s\n", loadCart("old.bin", restored, 10) < 0 ? "rejected" : "loaded");
    return 0;
}

@ steps
- نقرأ الحرف الأول ثم نعد التكرارات حتى يتغير الحرف أو يبلغ العداد 255.
- فك الضغط: لكل زوج نكتب الحرف `count` مرة.
@ flow

@ code
#include <stdio.h>

long encodeFile(const char *source, const char *target) {
    FILE *input = fopen(source, "rb"), *output = fopen(target, "wb");
    int ch, prev = fgetc(input), run = 1;
    long written = 0;
    while (prev != EOF) {
        ch = fgetc(input);
        if (ch == prev && run < 255) {
            run++;
        } else {
            fputc(run, output);                     // (العدد، الحرف)
            fputc(prev, output);
            written += 2;
            prev = ch;
            run = 1;
        }
    }
    fclose(input);
    fclose(output);
    return written;
}

long decodeFile(const char *source, const char *target) {
    FILE *input = fopen(source, "rb"), *output = fopen(target, "wb");
    int run, ch, rep;
    long written = 0;
    while ((run = fgetc(input)) != EOF && (ch = fgetc(input)) != EOF)
        for (rep = 0; rep < run; rep++) { fputc(ch, output); written++; }
    fclose(input);
    fclose(output);
    return written;
}

int main(void) {
    FILE *file = fopen("status.txt", "wb");
    int idx;
    for (idx = 0; idx < 300; idx++) fputc(idx < 120 ? 'A' : idx < 125 ? 'B' : idx < 290 ? 'A' : 'C', file);
    fclose(file);
    printf("encoded size: %ld bytes (original 300)\n", encodeFile("status.txt", "status.rle"));
    printf("decoded size: %ld bytes\n", decodeFile("status.rle", "status.out"));
    file = fopen("status.rle", "rb");
    printf("pairs:");
    while ((idx = fgetc(file)) != EOF) printf(" %d%c", idx, fgetc(file));
    printf("\n");
    fclose(file);
    return 0;
}

#include <stdio.h>

int isNumber(const char *text) {
    int dots = 0, digits = 0;
    for (; *text != '\0'; text++) {
        if (*text == '.') dots++;
        else if (*text >= '0' && *text <= '9') digits++;
        else return 0;
    }
    return digits > 0 && dots <= 1;
}

int main(void) {
    FILE *input = fopen("raw.csv", "w"), *clean, *rejected;
    char line[100], copy[100], *fields[8], *reason;
    int pos, count, idx, kept = 0, dropped = 0, ch;
    fprintf(input, "7781,Amina,Oran,12500\n7782,Karim,,3200\n7783,Lina,Setif,89x0\n");

        for (pos = 0; (copy[pos] = line[pos]) != '\0'; pos++) {}   // نسخة قبل التقسيم

        for (idx = 0; idx < count && reason == NULL; idx++) if (fields[idx][0] == '\0') reason = "empty field";

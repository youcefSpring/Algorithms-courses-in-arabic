\begin{itemize}
    \item \textbf{الجدول}: مجموعة عناصر \textbf{من نفس النوع} متجاورة في الذاكرة، لها اسم واحد، ونصل إلى كل عنصر بـ\textbf{فهرسه}.
    \item الفهرس في \en{C} يبدأ من \en{0} وينتهي عند \code{SIZE - 1}.
    \item \textbf{التصريح}: \code{double prices[50];} و\textbf{التهيئة}: \code{int stock[4] = \{5, 0, 12, 7\};}
    \item \textbf{الحجم المستعمل} \code{nbItems} قد يكون أصغر من \textbf{السعة} \code{MAX}.
\end{itemize}

\begin{tbox}{\faLightbulb\quad تعريفات}
\begin{itemize}
    \item \textbf{الملف}: بيانات تبقى بعد انتهاء البرنامج. نصل إليه عبر مؤشر \code{FILE *file = fopen(path, mode);}
    \item \textbf{الأوضاع}: \code{"r"} قراءة، \code{"w"} كتابة (يمسح المحتوى)، \code{"a"} إضافة في النهاية، \code{"r+"} قراءة وكتابة، ونضيف \code{b} للملفات الثنائية.
    \item \textbf{ملف نصي}: أسطر مقروءة (\code{fprintf}، \code{fscanf}، \code{fgets}). \textbf{ملف ثنائي}: نسخة من الذاكرة (\code{fwrite}، \code{fread}).
    \item \textbf{الموضع}: \code{fseek} للقفز، \code{ftell} لمعرفة الموضع، \code{rewind} للعودة إلى البداية.
\end{itemize}
\end{tbox}

\begin{obox}{\faExclamationTriangle\quad برامج هذا الفصل مستقلة}
كل برنامج ينشئ ملفاته بنفسه ثم يقرؤها، لذلك يمكن تنفيذه مباشرة وإعادة إنتاج نتائجه.
لا تنسَ \code{fclose}: البيانات المكتوبة قد تبقى في الذاكرة المؤقتة ولا تصل إلى القرص قبل الإغلاق.
\end{obox}

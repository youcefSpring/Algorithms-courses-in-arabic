\begin{tbox}{\faLightbulb\quad تعريفات}
\begin{itemize}
    \item \textbf{السلسلة النصية} في \en{C}: جدول من \code{char} ينتهي بالحرف الخاص \code{'\textbackslash 0'}.
    \item \textbf{التصريح}: \code{char name[20] = "Laptop";} يحجز \en{20} خانة، منها \en{7} مستعملة (\en{6} حروف + \code{'\textbackslash 0'}).
    \item \textbf{الطول} هو عدد الحروف قبل \code{'\textbackslash 0'}، ونحسبه بحلقة.
    \item كل حرف عدد صحيح (رمز \en{ASCII}): \code{'A'} = \en{65}، \code{'a'} = \en{97}، \code{'0'} = \en{48}.
\end{itemize}
\end{tbox}
